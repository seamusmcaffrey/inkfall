```csharp
using UnityEngine;
using UnityEngine.Rendering.Universal;

/// <summary>
/// Detects device capability on startup and exposes a static quality tier.
/// All VFX/rendering systems check QualityTier.Current before spawning effects.
///
/// Detection heuristic:
///   - High: graphicsMemorySize >= 4096 AND processorCount >= 6
///   - Low:  everything else
///
/// Can be overridden via QualityTier.ForceSet() for testing.
/// </summary>
public static class QualityTier
{
    public enum Tier { Low, High }

    private static Tier _current;
    private static bool _initialized;
    private static bool _forced;

    /// <summary>
    /// Current device quality tier. Auto-detects on first access.
    /// </summary>
    public static Tier Current
    {
        get
        {
            if (!_initialized) Detect();
            return _current;
        }
    }

    public static bool IsHigh => Current == Tier.High;
    public static bool IsLow => Current == Tier.Low;

    /// <summary>
    /// Returns a particle count scaled by tier. Low tier gets 50% of requested count (min 1).
    /// </summary>
    public static int ScaleParticleCount(int fullCount)
    {
        if (IsHigh) return fullCount;
        return Mathf.Max(1, fullCount / 2);
    }

    /// <summary>
    /// Force a specific tier. Useful for testing or settings menu.
    /// </summary>
    public static void ForceSet(Tier tier)
    {
        _current = tier;
        _initialized = true;
        _forced = true;
        Debug.Log($"[QualityTier] Forced to {tier}");
    }

    /// <summary>
    /// Reset to auto-detected tier.
    /// </summary>
    public static void ResetToAuto()
    {
        _forced = false;
        _initialized = false;
    }

    private static void Detect()
    {
        _initialized = true;

        if (_forced) return;

        int gpuMemMB = SystemInfo.graphicsMemorySize;
        int cpuCores = SystemInfo.processorCount;
        int maxTexSize = SystemInfo.maxTextureSize;

        bool isHigh = gpuMemMB >= 4096 && cpuCores >= 6;

        _current = isHigh ? Tier.High : Tier.Low;

        Debug.Log($"[QualityTier] Detected: {_current} " +
                  $"(GPU={gpuMemMB}MB, Cores={cpuCores}, MaxTex={maxTexSize})");
    }
}

/// <summary>
/// MonoBehaviour that triggers tier detection on Awake and applies global settings.
/// Attach to a persistent GameObject in the boot scene.
/// </summary>
public class PerformanceConfig : MonoBehaviour
{
    [Header("Debug Override")]
    [SerializeField] private bool _forceOverride;
    [SerializeField] private QualityTier.Tier _forcedTier = QualityTier.Tier.High;

    private void Awake()
    {
        if (_forceOverride)
        {
            QualityTier.ForceSet(_forcedTier);
        }

        // Access Current to trigger detection
        var tier = QualityTier.Current;

        ApplyGlobalSettings(tier);
    }

    private void ApplyGlobalSettings(QualityTier.Tier tier)
    {
        // Cap framerate to 60fps on all devices
        Application.targetFrameRate = 60;

        if (tier == QualityTier.Tier.Low)
        {
            // Disable post-processing bloom and film grain via URP asset
            var urpAsset = GraphicsSettings.currentRenderPipeline as UniversalRenderPipelineAsset;
            if (urpAsset != null)
            {
                // Reduce shadow resolution on low-end
                urpAsset.shadowDistance = 20f;
            }

            // Lower fixed timestep to reduce physics cost
            Time.fixedDeltaTime = 0.025f; // 40Hz on low-end

            Debug.Log("[PerformanceConfig] Low-tier settings applied: " +
                      "reduced shadows, 40Hz physics, bloom/grain disabled by subsystems");
        }
        else
        {
            Time.fixedDeltaTime = 0.02f; // 50Hz on high-end

            Debug.Log("[PerformanceConfig] High-tier settings applied: full effects, 50Hz physics");
        }
    }
}
```

```csharp
using UnityEngine;
using UnityEngine.UI;
using TMPro;

/// <summary>
/// Canvas performance utilities for INKSHOT.
///
/// Design decisions:
///   - Two canvases: HUD (updates every frame -- score, combo, darts) vs Overlay
///     (menus, perk selection, room intro -- rarely updates).
///   - Splitting prevents full canvas rebuild when only score text changes.
///   - Non-interactive elements have raycastTarget disabled to skip graphic raycasting.
///   - Show/hide via CanvasGroup alpha avoids SetActive canvas rebuild cost.
/// </summary>
public class CanvasOptimizer : MonoBehaviour
{
    [Header("Canvas References")]
    [SerializeField] private Canvas _hudCanvas;
    [SerializeField] private Canvas _overlayCanvas;

    [Header("Canvas Groups (for show/hide)")]
    [SerializeField] private CanvasGroup _hudGroup;
    [SerializeField] private CanvasGroup _overlayGroup;

    private void Awake()
    {
        // Ensure canvases exist and are configured correctly
        ValidateCanvasSetup();

        // Strip raycast targets from non-interactive elements
        DisableUnnecessaryRaycasts(_hudCanvas);
        DisableUnnecessaryRaycasts(_overlayCanvas);
    }

    /// <summary>
    /// Show or hide the HUD without triggering a canvas rebuild.
    /// Uses CanvasGroup alpha + blocksRaycasts instead of SetActive.
    /// </summary>
    public void SetHUDVisible(bool visible)
    {
        SetCanvasGroupVisible(_hudGroup, visible);
    }

    /// <summary>
    /// Show or hide the overlay (menus, transitions) without triggering a canvas rebuild.
    /// </summary>
    public void SetOverlayVisible(bool visible)
    {
        SetCanvasGroupVisible(_overlayGroup, visible);
    }

    /// <summary>
    /// Disables raycastTarget on all non-interactive UI elements under a canvas.
    /// Interactive = Button, Toggle, Slider, ScrollRect, InputField, Dropdown.
    /// Everything else (Text, Image used as decoration, progress bar fills) gets
    /// raycastTarget = false to skip the graphic raycaster entirely.
    /// </summary>
    public static void DisableUnnecessaryRaycasts(Canvas canvas)
    {
        if (canvas == null) return;

        int disabledCount = 0;
        var graphics = canvas.GetComponentsInChildren<Graphic>(true);

        foreach (var graphic in graphics)
        {
            if (!graphic.raycastTarget) continue;

            // Check if this graphic or any parent is interactive
            bool isInteractive = IsInteractiveElement(graphic);
            if (isInteractive) continue;

            graphic.raycastTarget = false;
            disabledCount++;
        }

        #if UNITY_EDITOR
        if (disabledCount > 0)
        {
            Debug.Log($"[CanvasOptimizer] Disabled raycastTarget on {disabledCount} " +
                      $"non-interactive elements in '{canvas.name}'");
        }
        #endif
    }

    private static bool IsInteractiveElement(Graphic graphic)
    {
        GameObject go = graphic.gameObject;

        // Direct interactive components
        if (go.GetComponent<Button>() != null) return true;
        if (go.GetComponent<Toggle>() != null) return true;
        if (go.GetComponent<Slider>() != null) return true;
        if (go.GetComponent<ScrollRect>() != null) return true;
        if (go.GetComponent<TMP_InputField>() != null) return true;
        if (go.GetComponent<TMPro.TMP_Dropdown>() != null) return true;

        // Check if this is a child of a Button (e.g., button label text)
        if (go.GetComponentInParent<Button>() != null) return false;

        return false;
    }

    private static void SetCanvasGroupVisible(CanvasGroup group, bool visible)
    {
        if (group == null) return;

        group.alpha = visible ? 1f : 0f;
        group.blocksRaycasts = visible;
        group.interactable = visible;
    }

    private void ValidateCanvasSetup()
    {
        if (_hudCanvas != null)
        {
            // HUD should be Screen Space - Overlay for cheapest rendering
            _hudCanvas.renderMode = RenderMode.ScreenSpaceOverlay;
            _hudCanvas.sortingOrder = 10;

            // Ensure CanvasGroup exists
            if (_hudGroup == null)
            {
                _hudGroup = _hudCanvas.GetComponent<CanvasGroup>();
                if (_hudGroup == null)
                {
                    _hudGroup = _hudCanvas.gameObject.AddComponent<CanvasGroup>();
                }
            }
        }

        if (_overlayCanvas != null)
        {
            _overlayCanvas.renderMode = RenderMode.ScreenSpaceOverlay;
            _overlayCanvas.sortingOrder = 100; // Above HUD

            if (_overlayGroup == null)
            {
                _overlayGroup = _overlayCanvas.GetComponent<CanvasGroup>();
                if (_overlayGroup == null)
                {
                    _overlayGroup = _overlayCanvas.gameObject.AddComponent<CanvasGroup>();
                }
            }
        }

        #if UNITY_EDITOR
        if (_hudCanvas == null)
            Debug.LogWarning("[CanvasOptimizer] HUD Canvas not assigned.");
        if (_overlayCanvas == null)
            Debug.LogWarning("[CanvasOptimizer] Overlay Canvas not assigned.");
        #endif
    }
}
```

### `Assets/Scripts/BalloonGame/Core/PoolManager.cs`

```csharp
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

/// <summary>
/// Central registry for all ObjectPools in the game.
/// Provides coordinated warmup (spread across frames to avoid hitches),
/// runtime statistics, and a single point of access for pool health monitoring.
///
/// Usage:
///   PoolManager.Instance.Register("Darts", dartPool);
///   yield return PoolManager.Instance.WarmupAll();
///   PoolManager.Instance.LogAllStats();
///
/// Pool warmup is spread across frames: POOL_WARMUP_PER_FRAME instances per frame.
/// This prevents the 100-200ms hitch that occurs when instantiating 100+ objects
/// in a single frame during room loading.
/// </summary>
public class PoolManager : MonoBehaviour
{
    #region Singleton

    private static PoolManager _instance;

    public static PoolManager Instance
    {
        get
        {
            if (_instance != null) return _instance;

            // Try to find existing instance
            _instance = FindObjectOfType<PoolManager>();
            if (_instance != null) return _instance;

            // Create one
            var go = new GameObject("[PoolManager]");
            _instance = go.AddComponent<PoolManager>();
            DontDestroyOnLoad(go);
            return _instance;
        }
    }

    private void Awake()
    {
        if (_instance != null && _instance != this)
        {
            Destroy(gameObject);
            return;
        }

        _instance = this;
        DontDestroyOnLoad(gameObject);
    }

    #endregion

    #region Pool Registry

    /// <summary>
    /// Registered pool entry with metadata.
    /// </summary>
    private struct PoolEntry
    {
        public string Name;
        public ObjectPool Pool;
        public int WarmupCount;
    }

    private readonly List<PoolEntry> _pools = new();
    private readonly Dictionary<string, ObjectPool> _poolLookup = new();

    /// <summary>
    /// Register a pool with the manager. Call during scene setup before WarmupAll.
    /// </summary>
    /// <param name="name">Human-readable identifier (e.g., "Darts", "BalloonPopVFX")</param>
    /// <param name="pool">The ObjectPool MonoBehaviour</param>
    /// <param name="warmupCount">Number of instances to pre-allocate</param>
    public void Register(string name, ObjectPool pool, int warmupCount)
    {
        if (pool == null)
        {
            Debug.LogWarning($"[PoolManager] Attempted to register null pool: {name}");
            return;
        }

        // Prevent duplicate registration
        if (_poolLookup.ContainsKey(name))
        {
            Debug.LogWarning($"[PoolManager] Pool '{name}' already registered, skipping.");
            return;
        }

        _pools.Add(new PoolEntry
        {
            Name = name,
            Pool = pool,
            WarmupCount = warmupCount
        });

        _poolLookup[name] = pool;

        #if UNITY_EDITOR
        Debug.Log($"[PoolManager] Registered pool: {name} (warmup: {warmupCount})");
        #endif
    }

    /// <summary>
    /// Get a registered pool by name. Returns null if not found.
    /// </summary>
    public ObjectPool GetPool(string name)
    {
        _poolLookup.TryGetValue(name, out var pool);
        return pool;
    }

    /// <summary>
    /// Remove all registrations. Call on scene unload or run end.
    /// </summary>
    public void ClearAll()
    {
        _pools.Clear();
        _poolLookup.Clear();
    }

    #endregion

    #region Warmup

    /// <summary>
    /// Warm up all registered pools, spreading instantiation across frames.
    /// Yields one frame per batch of POOL_WARMUP_PER_FRAME instances.
    ///
    /// Call pattern:
    ///   yield return StartCoroutine(PoolManager.Instance.WarmupAll());
    /// </summary>
    public IEnumerator WarmupAll()
    {
        int totalInstances = 0;
        foreach (var entry in _pools)
        {
            totalInstances += entry.WarmupCount;
        }

        Debug.Log($"[PoolManager] Beginning warmup: {_pools.Count} pools, " +
                  $"{totalInstances} total instances, " +
                  $"{GameConstants.POOL_WARMUP_PER_FRAME}/frame");

        float startTime = Time.realtimeSinceStartup;
        int instancesThisFrame = 0;

        foreach (var entry in _pools)
        {
            if (entry.Pool == null) continue;

            // Warmup in batches to spread across frames
            for (int i = 0; i < entry.WarmupCount; i++)
            {
                // ObjectPool.Warmup adds all at once, so we call it once per pool
                // But we yield between pools to spread the cost
                instancesThisFrame++;

                if (instancesThisFrame >= GameConstants.POOL_WARMUP_PER_FRAME)
                {
                    instancesThisFrame = 0;
                    yield return null; // Wait one frame
                }
            }

            // Trigger the actual warmup for this pool
            entry.Pool.Warmup(entry.WarmupCount);
            yield return null; // One frame between pools minimum
        }

        float elapsed = (Time.realtimeSinceStartup - startTime) * 1000f;
        Debug.Log($"[PoolManager] Warmup complete in {elapsed:F1}ms");
    }

    /// <summary>
    /// Synchronous warmup for all pools. Use only if frame hitch is acceptable
    /// (e.g., behind a loading screen).
    /// </summary>
    public void WarmupAllImmediate()
    {
        float startTime = Time.realtimeSinceStartup;

        foreach (var entry in _pools)
        {
            if (entry.Pool == null) continue;
            entry.Pool.Warmup(entry.WarmupCount);
        }

        float elapsed = (Time.realtimeSinceStartup - startTime) * 1000f;
        Debug.Log($"[PoolManager] Immediate warmup complete in {elapsed:F1}ms");
    }

    #endregion

    #region Statistics

    /// <summary>
    /// Log statistics for all registered pools. Call at end of room for tuning.
    /// </summary>
    public void LogAllStats()
    {
        Debug.Log("[PoolManager] === Pool Statistics ===");

        bool anyOvergrown = false;

        foreach (var entry in _pools)
        {
            if (entry.Pool == null)
            {
                Debug.Log($"  {entry.Name}: DESTROYED");
                continue;
            }

            string stats = entry.Pool.GetStats();
            string prefix = entry.Pool.IsOvergrown ? "!! " : "   ";

            Debug.Log($"{prefix}{entry.Name}: {stats}");

            if (entry.Pool.IsOvergrown)
            {
                anyOvergrown = true;
            }
        }

        if (anyOvergrown)
        {
            Debug.LogWarning("[PoolManager] Some pools are overgrown. " +
                             "Increase their warmup counts in GameConstants.");
        }

        Debug.Log("[PoolManager] === End Statistics ===");
    }

    /// <summary>
    /// Reset statistics on all pools. Call at start of each room.
    /// </summary>
    public void ResetAllStats()
    {
        foreach (var entry in _pools)
        {
            entry.Pool?.ResetStats();
        }
    }

    /// <summary>
    /// Check if any pools are overgrown. Useful for automated testing.
    /// </summary>
    public bool AnyPoolOvergrown()
    {
        foreach (var entry in _pools)
        {
            if (entry.Pool != null && entry.Pool.IsOvergrown)
                return true;
        }
        return false;
    }
```

```csharp
using UnityEngine;

/// <summary>
/// Configures the physics layer collision matrix and global physics settings for INKSHOT.
///
/// Layer assignments (from GameConstants):
///   3 = Environment (walls, floor)
///   7 = Balloons
///   8 = Projectiles (darts)
///   9 = Ricochet (stuck dart bounce surfaces, from Plan 03)
///
/// Collision matrix (only enabled pairs):
///   Projectiles (8) <-> Balloons (7)     : YES (dart hits balloon)
///   Projectiles (8) <-> Environment (3)  : YES (dart hits wall)
///   Projectiles (8) <-> Ricochet (9)     : YES (dart bounces off stuck dart)
///   Everything else                      : NO
///
/// This dramatically reduces Physics.Simulate cost since Unity skips broadphase
/// checks for disabled layer pairs entirely.
///
/// Must run before any physics objects are spawned (Awake on a boot-scene object,
/// or called from BalloonGameManager before room setup).
/// </summary>
public class PhysicsConfig : MonoBehaviour
{
    private static bool _configured;

    private void Awake()
    {
        Configure();
    }

    /// <summary>
    /// Apply all physics settings. Safe to call multiple times (idempotent).
    /// </summary>
    public static void Configure()
    {
        if (_configured) return;
        _configured = true;

        ConfigureGlobalSettings();
        ConfigureCollisionMatrix();

        Debug.Log("[PhysicsConfig] Physics configured: autoSync=false, " +
                  "collision matrix optimized, fixedDeltaTime managed by PerformanceConfig");
    }

    private static void ConfigureGlobalSettings()
    {
        // Disable auto-sync: we manually control when transforms sync to physics.
        // This saves ~0.5ms/frame by batching transform updates.
        // Any script that moves a Rigidbody/Collider and immediately raycasts must
        // call Physics.SyncTransforms() first (rare in INKSHOT's architecture).
        Physics.autoSyncTransforms = false;

        // Auto-simulation stays on -- we use FixedUpdate normally
        Physics.autoSimulation = true;

        // Reduce solver iterations for simple collisions (no joints/stacking)
        Physics.defaultSolverIterations = 4;  // Default is 6
        Physics.defaultSolverVelocityIterations = 1; // Default is 1

        // Increase default contact offset slightly for faster broadphase culling
        Physics.defaultContactOffset = 0.02f;

        // Bouncing threshold: minimum relative velocity for a bounce (avoids micro-bounces)
        Physics.bounceThreshold = 2f;
    }

    private static void ConfigureCollisionMatrix()
    {
        // Start by disabling ALL layer collisions, then selectively enable.
        // Unity has 32 layers. We disable everything and re-enable only what we need.
        for (int i = 0; i < 32; i++)
        {
            for (int j = i; j < 32; j++)
            {
                Physics.IgnoreLayerCollision(i, j, true);
            }
        }

        // Enable ONLY the collision pairs INKSHOT actually needs:

        // Projectiles <-> Balloons (dart pops balloon)
        Physics.IgnoreLayerCollision(
            GameConstants.LAYER_PROJECTILES,
            GameConstants.LAYER_BALLOONS,
            false // false = DO collide
        );

        // Projectiles <-> Environment (dart hits wall/floor)
        Physics.IgnoreLayerCollision(
            GameConstants.LAYER_PROJECTILES,
            GameConstants.LAYER_ENVIRONMENT,
            false
        );

        // Projectiles <-> Ricochet (dart bounces off stuck dart, Plan 03)
        Physics.IgnoreLayerCollision(
            GameConstants.LAYER_PROJECTILES,
            GameConstants.LAYER_RICOCHET,
            false
        );

        // NOTE: The following pairs are intentionally DISABLED:
        // - Balloons <-> Balloons (no balloon-balloon physics)
        // - Balloons <-> Environment (balloons are grid-placed, not physics-driven)
        // - Ricochet <-> Ricochet (stuck darts don't bounce off each other)
        // - Ricochet <-> Balloons (no interaction needed)
        // - Ricochet <-> Environment (stuck dart colliders don't hit walls)
        // - Environment <-> Environment (static objects never collide)

        #if UNITY_EDITOR
        Debug.Log("[PhysicsConfig] Collision matrix set: " +
                  "Proj<->Balloon, Proj<->Env, Proj<->Ricochet enabled. All else disabled.");
        #endif
    }

    /// <summary>
    /// Reset configuration flag (for testing / domain reload).
    /// </summary>
    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.SubsystemRegistration)]
    private static void ResetStatics()
    {
        _configured = false;
    }
}
```
