```csharp
// Assets/Scripts/BalloonGame/SlingshotInput.cs
using System;
using UnityEngine;
using UnityEngine.InputSystem;
using UnityEngine.InputSystem.EnhancedTouch;
using Touch = UnityEngine.InputSystem.EnhancedTouch.Touch;

public class SlingshotInput : MonoBehaviour
{
    public event Action<Vector3> OnLaunch;
    public event Action<Vector2> OnPullUpdate;
    public event Action OnPullCancel;
    public event Action OnPullStart;

    private Camera _camera;
    private bool _isAiming;
    private bool _isPullArmed;
    private bool _hasClearedDeadZone;
    private bool _canFire = true;
    private Vector2 _pullVector;
    private int _activeFingerIndex = -1;

    public bool IsAiming => _isAiming;
    public bool IsPullArmed => _isPullArmed;

    public float PullStrength
    {
        get
        {
            float distance = _pullVector.magnitude;
            return Mathf.Pow(
                Mathf.Clamp01(distance / GameConstants.MAX_PULL_DISTANCE),
                GameConstants.PULL_SPEED_EXPONENT);
        }
    }

    public Vector3 PreviewVelocity
    {
        get
        {
            float distance = _pullVector.magnitude;
            if (distance < 0.05f)
            {
                return Vector3.zero;
            }

            Vector2 direction = -_pullVector.normalized;
            float power = Mathf.Pow(
                Mathf.Clamp01(distance / GameConstants.MAX_PULL_DISTANCE),
                GameConstants.PULL_SPEED_EXPONENT);
            float speed = GameConstants.MIN_LAUNCH_SPEED +
                          power * (GameConstants.MAX_LAUNCH_SPEED - GameConstants.MIN_LAUNCH_SPEED);
            return new Vector3(direction.x, direction.y, 0f) * speed;
        }
    }

    private void OnEnable()
    {
        EnhancedTouchSupport.Enable();
    }

    private void OnDisable()
    {
        EnhancedTouchSupport.Disable();
    }

    private void Update()
    {
        if (!_canFire)
        {
            return;
        }

        if (Touch.activeTouches.Count > 0)
        {
            HandleTouchInput();
        }
        else
        {
            HandlePointerInput();
        }
    }

    public void SetCanFire(bool canFire)
    {
        _canFire = canFire;
        if (!canFire && _isAiming)
        {
            CancelPull();
        }
    }

    private void HandleTouchInput()
    {
        // If not aiming, look for a new touch that starts in the activation zone
        if (!_isAiming)
        {
            foreach (var touch in Touch.activeTouches)
            {
                if (touch.phase == UnityEngine.InputSystem.TouchPhase.Began)
                {
                    Vector2 worldPosition = ScreenToWorld(touch.screenPosition);
                    Vector2 launchPosition = new(
                        GameConstants.LAUNCH_POSITION.x,
                        GameConstants.LAUNCH_POSITION.y);

                    if (Vector2.Distance(worldPosition, launchPosition) <=
                        GameConstants.AIM_ACTIVATION_RADIUS)
                    {
                        _activeFingerIndex = touch.finger.index;
                        BeginPullAtPosition(worldPosition);
                        break;
                    }
                }
            }

            return;
        }

        // Already aiming — find our tracked finger
        Touch? trackedTouch = null;
        foreach (var touch in Touch.activeTouches)
        {
            if (touch.finger.index == _activeFingerIndex)
            {
                trackedTouch = touch;
                break;
            }
        }

        // Tracked finger lost — cancel
        if (trackedTouch == null)
        {
            CancelPull();
            return;
        }

        var tracked = trackedTouch.Value;

        if (tracked.phase == UnityEngine.InputSystem.TouchPhase.Ended ||
            tracked.phase == UnityEngine.InputSystem.TouchPhase.Canceled)
        {
            ReleasePull();
        }
        else if (tracked.phase == UnityEngine.InputSystem.TouchPhase.Moved ||
                 tracked.phase == UnityEngine.InputSystem.TouchPhase.Stationary)
        {
            UpdatePullAtPosition(ScreenToWorld(tracked.screenPosition));
        }
    }

    private void HandlePointerInput()
    {
        Pointer pointer = Pointer.current;
        if (pointer == null)
        {
            return;
        }

        if (pointer.press.wasPressedThisFrame)
        {
            Vector2 worldPosition = ScreenToWorld(pointer.position.ReadValue());
            Vector2 launchPosition = new(
                GameConstants.LAUNCH_POSITION.x,
                GameConstants.LAUNCH_POSITION.y);

            if (Vector2.Distance(worldPosition, launchPosition) <=
                GameConstants.AIM_ACTIVATION_RADIUS)
            {
                _activeFingerIndex = -1;
                BeginPullAtPosition(worldPosition);
            }
        }
        else if (pointer.press.isPressed && _isAiming)
        {
            UpdatePullAtPosition(ScreenToWorld(pointer.position.ReadValue()));
        }
        else if (pointer.press.wasReleasedThisFrame && _isAiming)
        {
            ReleasePull();
        }
    }

    private void BeginPullAtPosition(Vector2 worldPosition)
    {
        _isAiming = true;
        _isPullArmed = false;
        _hasClearedDeadZone = false;
        _pullVector = Vector2.zero;

        OnPullStart?.Invoke();
        OnPullUpdate?.Invoke(_pullVector);
    }

    private void UpdatePullAtPosition(Vector2 worldPosition)
    {
        Vector2 launchPosition = new(
            GameConstants.LAUNCH_POSITION.x,
            GameConstants.LAUNCH_POSITION.y);
        Vector2 rawPull = worldPosition - launchPosition;
        float rawDistance = rawPull.magnitude;

        // Cancel-by-return: if player drags back to origin, cancel the pull
        if (_hasClearedDeadZone && rawDistance <= GameConstants.PULL_CANCEL_RETURN_RADIUS)
        {
            CancelPull();
            return;
        }

        // Dead zone: don't register pull until past dead zone threshold
        if (!_hasClearedDeadZone)
        {
            if (rawDistance < GameConstants.PULL_DEAD_ZONE)
            {
                return;
            }

            _hasClearedDeadZone = true;
        }

        float distance = Mathf.Min(rawDistance, GameConstants.MAX_PULL_DISTANCE);
        _pullVector = rawPull.sqrMagnitude > Mathf.Epsilon
            ? rawPull.normalized * distance
            : Vector2.zero;

        // Arm the pull once minimum distance is reached
        _isPullArmed = distance >= GameConstants.MIN_PULL_DISTANCE;

        OnPullUpdate?.Invoke(_pullVector);
    }

    private void ReleasePull()
    {
        _isAiming = false;
        _activeFingerIndex = -1;

        if (!_isPullArmed)
        {
            _pullVector = Vector2.zero;
            _isPullArmed = false;
            OnPullCancel?.Invoke();
            return;
        }

        Vector3 velocity = PreviewVelocity;
        _pullVector = Vector2.zero;
        _isPullArmed = false;

        if (velocity.sqrMagnitude > 1f)
        {
            OnLaunch?.Invoke(velocity);
        }
        else
        {
            OnPullCancel?.Invoke();
        }
    }

    private void CancelPull()
    {
        _isAiming = false;
        _isPullArmed = false;
        _hasClearedDeadZone = false;
        _activeFingerIndex = -1;
        _pullVector = Vector2.zero;
        OnPullCancel?.Invoke();
    }

    private Vector2 ScreenToWorld(Vector2 screenPosition)
    {
        if (_camera == null)
        {
            _camera = Camera.main;
        }

        Vector3 worldPosition = _camera.ScreenToWorldPoint(
            new Vector3(screenPosition.x, screenPosition.y,
                Mathf.Abs(_camera.transform.position.z)));
        return new Vector2(worldPosition.x, worldPosition.y);
    }
}
```

```csharp
// Assets/Scripts/BalloonGame/SlingshotVisuals.cs
using UnityEngine;

[RequireComponent(typeof(SlingshotInput))]
public class SlingshotVisuals : MonoBehaviour
{
    private SlingshotInput _input;
    private LineRenderer _bandLine;
    private GameObject[] _trajectoryDots;
    private Material _dotMaterial;
    private GameObject _originIndicator;
    private Material _originMaterial;

    private void Awake()
    {
        _input = GetComponent<SlingshotInput>();
        _input.OnPullStart += OnPullStart;
        _input.OnPullUpdate += OnPullUpdate;
        _input.OnPullCancel += HideAll;

        _bandLine = CreateBandLine();
        _trajectoryDots = CreateTrajectoryDots();
        _originIndicator = CreateOriginIndicator();

        HideAll();
    }

    private void OnDestroy()
    {
        if (_input == null)
        {
            return;
        }

        _input.OnPullStart -= OnPullStart;
        _input.OnPullUpdate -= OnPullUpdate;
        _input.OnPullCancel -= HideAll;
    }

    private void OnPullStart()
    {
        _originIndicator.SetActive(true);
    }

    private void OnPullUpdate(Vector2 pullVector)
    {
        Vector3 origin = GameConstants.LAUNCH_POSITION;
        Vector3 pullEnd = origin + new Vector3(pullVector.x, pullVector.y, 0f);

        // Rubber band
        _bandLine.positionCount = 2;
        _bandLine.SetPosition(0, origin + Vector3.back * 0.1f);
        _bandLine.SetPosition(1, pullEnd + Vector3.back * 0.1f);

        // Trajectory dots
        Vector3 velocity = _input.PreviewVelocity;
        if (velocity.sqrMagnitude < 1f)
        {
            HideTrajectoryDots();
            return;
        }

        Vector3 gravity = Physics.gravity;
        float time = Time.time;

        for (int i = 0; i < _trajectoryDots.Length; i++)
        {
            float t = (float)i / (_trajectoryDots.Length - 1) * GameConstants.TRAJECTORY_DURATION;
            Vector3 position = origin + velocity * t + 0.5f * gravity * t * t;
            _trajectoryDots[i].transform.position = position + Vector3.back * 0.1f;
            _trajectoryDots[i].SetActive(true);

            // Pulse alpha: sine wave offset by index for wave effect
            float pulse = Mathf.Sin(
                time * GameConstants.TRAJECTORY_PULSE_SPEED + i * 0.4f);
            float pulseAlpha = 1f + pulse * GameConstants.TRAJECTORY_PULSE_AMPLITUDE;

            // Fade out toward end of trajectory
            float lifeFraction = 1f - ((float)i / _trajectoryDots.Length);
            float finalAlpha = Mathf.Clamp01(lifeFraction * pulseAlpha) * 0.7f;

            var renderer = _trajectoryDots[i].GetComponent<Renderer>();
            var block = new MaterialPropertyBlock();
            renderer.GetPropertyBlock(block);
            block.SetColor("_Color", new Color(1f, 1f, 1f, finalAlpha));
            renderer.SetPropertyBlock(block);
        }
    }

    private void Update()
    {
        // Pulse origin indicator
        if (_originIndicator.activeSelf)
        {
            float pulse = Mathf.Sin(Time.time * GameConstants.ORIGIN_INDICATOR_PULSE_SPEED);
            float alpha = Mathf.Lerp(
                GameConstants.ORIGIN_INDICATOR_MIN_ALPHA,
                GameConstants.ORIGIN_INDICATOR_MAX_ALPHA,
                (pulse + 1f) * 0.5f);

            var renderer = _originIndicator.GetComponent<Renderer>();
            var block = new MaterialPropertyBlock();
            renderer.GetPropertyBlock(block);
            block.SetColor("_Color", new Color(1f, 0.85f, 0.6f, alpha));
            renderer.SetPropertyBlock(block);
        }

        // Hide everything when not aiming
        if (!_input.IsAiming)
        {
            HideAll();
        }
    }

    private void HideAll()
    {
        _bandLine.positionCount = 0;
        HideTrajectoryDots();
        _originIndicator.SetActive(false);
    }

    private void HideTrajectoryDots()
    {
        for (int i = 0; i < _trajectoryDots.Length; i++)
        {
            _trajectoryDots[i].SetActive(false);
        }
    }

    private LineRenderer CreateBandLine()
    {
        var lineObject = new GameObject("RubberBand");
        lineObject.transform.SetParent(transform, false);

        var lineRenderer = lineObject.AddComponent<LineRenderer>();

        // Emissive red material for glow effect
        var bandMaterial = new Material(Shader.Find("Sprites/Default"));
        Color bandColor = new Color(0.95f, 0.15f, 0.1f, 0.95f);
        bandMaterial.color = bandColor;
        bandMaterial.EnableKeyword("_EMISSION");
        bandMaterial.SetColor("_EmissionColor", bandColor * 0.5f);

        lineRenderer.material = bandMaterial;
        lineRenderer.startColor = bandColor;
        lineRenderer.endColor = new Color(bandColor.r, bandColor.g, bandColor.b, 0.7f);
        lineRenderer.startWidth = GameConstants.BAND_WIDTH;
        lineRenderer.endWidth = GameConstants.BAND_WIDTH * 0.8f;
        lineRenderer.positionCount = 0;
        lineRenderer.useWorldSpace = true;
        lineRenderer.numCapVertices = 4;

        return lineRenderer;
    }

    private GameObject[] CreateTrajectoryDots()
    {
        int count = GameConstants.TRAJECTORY_DOT_COUNT;
        var dots = new GameObject[count];

        _dotMaterial = new Material(Shader.Find("Sprites/Default"));
        _dotMaterial.color = new Color(1f, 1f, 1f, 0.6f);

        for (int i = 0; i < count; i++)
        {
            var dot = GameObject.CreatePrimitive(PrimitiveType.Quad);
            dot.name = $"TrajectoryDot_{i}";
            dot.transform.SetParent(transform, false);
            dot.transform.localScale = Vector3.one * GameConstants.TRAJECTORY_DOT_SIZE;

            // Remove collider — purely visual
            Object.Destroy(dot.GetComponent<Collider>());

            dot.GetComponent<Renderer>().material = _dotMaterial;
            dot.SetActive(false);
            dots[i] = dot;
        }

        return dots;
    }

    private GameObject CreateOriginIndicator()
    {
        var indicator = GameObject.CreatePrimitive(PrimitiveType.Quad);
        indicator.name = "LaunchOriginIndicator";
        indicator.transform.SetParent(transform, false);
        indicator.transform.position = GameConstants.LAUNCH_POSITION + Vector3.back * 0.2f;
        indicator.transform.localScale = Vector3.one * GameConstants.ORIGIN_INDICATOR_SIZE;

        Object.Destroy(indicator.GetComponent<Collider>());

        // Radial gradient material — warm glow circle
        _originMaterial = new Material(Shader.Find("Sprites/Default"));
        _originMaterial.color = new Color(1f, 0.85f, 0.6f, GameConstants.ORIGIN_INDICATOR_MIN_ALPHA);

        indicator.GetComponent<Renderer>().material = _originMaterial;
        indicator.SetActive(false);

        return indicator;
    }
}
```

```csharp
// Assets/Scripts/BalloonGame/Visuals/LaunchLaneVisuals.cs
using UnityEngine;

/// <summary>
/// Creates visual dressing for the launch lane: floor panel, guide lines,
/// dashed center line, and cosmetic remaining-darts display.
/// Attach to a persistent scene object (e.g., the game manager or a dedicated "Lane" object).
/// </summary>
public class LaunchLaneVisuals : MonoBehaviour
{
    private GameObject _laneFloor;
    private GameObject _leftLine;
    private GameObject _rightLine;
    private GameObject[] _centerDashes;
    private GameObject[] _dartDisplays;
    private int _dartsRemaining;

    private static Material _lineMaterial;
    private static Material _floorMaterial;

    private void Awake()
    {
        CreateLaneFloor();
        CreateLaneGuideLines();
        CreateCenterDashes();
        CreateDartDisplay(GameConstants.STARTING_DARTS);
    }

    /// <summary>
    /// Call when a dart is launched to hide one cosmetic dart from the display.
    /// </summary>
    public void OnDartLaunched()
    {
        if (_dartDisplays == null)
        {
            return;
        }

        _dartsRemaining--;

        for (int i = 0; i < _dartDisplays.Length; i++)
        {
            _dartDisplays[i].SetActive(i < _dartsRemaining);
        }
    }

    /// <summary>
    /// Reset the dart display (e.g., on new room).
    /// </summary>
    public void ResetDartDisplay(int dartCount)
    {
        if (_dartDisplays != null)
        {
            for (int i = 0; i < _dartDisplays.Length; i++)
            {
                Object.Destroy(_dartDisplays[i]);
            }
        }

        CreateDartDisplay(dartCount);
    }

    private void CreateLaneFloor()
    {
        _laneFloor = GameObject.CreatePrimitive(PrimitiveType.Quad);
        _laneFloor.name = "LaneFloor";
        _laneFloor.transform.SetParent(transform, false);
        Object.Destroy(_laneFloor.GetComponent<Collider>());

        float laneHeight = GameConstants.LANE_TOP - GameConstants.LANE_BOTTOM;
        float laneWidth = GameConstants.BOARD_RIGHT - GameConstants.BOARD_LEFT;
        float laneCenterY = (GameConstants.LANE_TOP + GameConstants.LANE_BOTTOM) * 0.5f;

        _laneFloor.transform.position = new Vector3(0f, laneCenterY, 0.5f);
        _laneFloor.transform.localScale = new Vector3(laneWidth, laneHeight, 1f);

        _floorMaterial = GetFloorMaterial();
        _laneFloor.GetComponent<Renderer>().material = _floorMaterial;
    }

    private void CreateLaneGuideLines()
    {
        float laneHalfWidth = (GameConstants.BOARD_RIGHT - GameConstants.BOARD_LEFT) * 0.45f;

        _leftLine = CreateLaneLine("LeftGuideLine",
            new Vector3(-laneHalfWidth, 0f, 0f));
        _rightLine = CreateLaneLine("RightGuideLine",
            new Vector3(laneHalfWidth, 0f, 0f));
    }

    private GameObject CreateLaneLine(string name, Vector3 xOffset)
    {
        var line = GameObject.CreatePrimitive(PrimitiveType.Quad);
        line.name = name;
        line.transform.SetParent(transform, false);
        Object.Destroy(line.GetComponent<Collider>());

        float laneHeight = GameConstants.LANE_TOP - GameConstants.LANE_BOTTOM;
        float laneCenterY = (GameConstants.LANE_TOP + GameConstants.LANE_BOTTOM) * 0.5f;

        line.transform.position = new Vector3(xOffset.x, laneCenterY, 0.4f);
        line.transform.localScale = new Vector3(
            GameConstants.LANE_LINE_WIDTH,
            laneHeight,
            1f);

        var material = GetLineMaterial();
        line.GetComponent<Renderer>().material = material;

        return line;
    }

    private void CreateCenterDashes()
    {
        float laneHeight = GameConstants.LANE_TOP - GameConstants.LANE_BOTTOM;
        float dashStep = GameConstants.LANE_CENTER_DASH_LENGTH + GameConstants.LANE_CENTER_DASH_GAP;
        int dashCount = Mathf.FloorToInt(laneHeight / dashStep);

        _centerDashes = new GameObject[dashCount];

        for (int i = 0; i < dashCount; i++)
        {
            var dash = GameObject.CreatePrimitive(PrimitiveType.Quad);
            dash.name = $"CenterDash_{i}";
            dash.transform.SetParent(transform, false);
            Object.Destroy(dash.GetComponent<Collider>());

            float y = GameConstants.LANE_BOTTOM +
                       dashStep * 0.5f +
                       i * dashStep;

            dash.transform.position = new Vector3(0f, y, 0.4f);
            dash.transform.localScale = new Vector3(
                GameConstants.LANE_LINE_WIDTH * 0.8f,
                GameConstants.LANE_CENTER_DASH_LENGTH,
                1f);

            dash.GetComponent<Renderer>().material = GetLineMaterial();
            _centerDashes[i] = dash;
        }
    }

    private void CreateDartDisplay(int dartCount)
    {
        _dartsRemaining = dartCount;
        _dartDisplays = new GameObject[dartCount];

        float totalWidth = (dartCount - 1) * GameConstants.DART_DISPLAY_SPACING;
        float startX = -totalWidth * 0.5f;

        for (int i = 0; i < dartCount; i++)
        {
            var dart = CreateCosmeticDart($"DartDisplay_{i}");
            dart.transform.position = new Vector3(
                startX + i * GameConstants.DART_DISPLAY_SPACING,
                GameConstants.DART_DISPLAY_Y,
                0f);
            dart.transform.localScale = Vector3.one * GameConstants.DART_DISPLAY_SCALE;

            // Point upward (ready to launch)
            dart.transform.rotation = Quaternion.Euler(0f, 0f, 90f);

            _dartDisplays[i] = dart;
        }
    }

    /// <summary>
    /// Creates a small cosmetic dart mesh for the remaining-darts display.
    /// Uses DartMeshGenerator from Plan 02 if available, otherwise falls back
    /// to a simple capsule+sphere primitive approximation.
    /// </summary>
    private GameObject CreateCosmeticDart(string name)
    {
        var root = new GameObject(name);
        root.transform.SetParent(transform, false);

        // Try DartMeshGenerator from Plan 02
        Mesh dartMesh = null;

#if false
        // Uncomment when Plan 02 DartMeshGenerator is implemented:
        // dartMesh = DartMeshGenerator.Generate();
#endif

        if (dartMesh != null)
        {
            var meshObj = new GameObject("Mesh");
            meshObj.transform.SetParent(root.transform, false);
            var meshFilter = meshObj.AddComponent<MeshFilter>();
            meshFilter.mesh = dartMesh;
            var meshRenderer = meshObj.AddComponent<MeshRenderer>();
            meshRenderer.material = GetDartDisplayMaterial();
            return root;
        }

        // Fallback: capsule + sphere primitive (matches DartLauncher pattern)
        var body = GameObject.CreatePrimitive(PrimitiveType.Capsule);
        body.name = "Body";
        body.transform.SetParent(root.transform, false);
        body.transform.localRotation = Quaternion.Euler(0f, 0f, 90f);
        body.transform.localScale = new Vector3(0.1f, 0.25f, 0.1f);
        Object.Destroy(body.GetComponent<Collider>());
        body.GetComponent<Renderer>().material = GetDartDisplayMaterial();

        var tip = GameObject.CreatePrimitive(PrimitiveType.Sphere);
```

```csharp
// Assets/Scripts/BalloonGame/AimAssist.cs
using UnityEngine;

/// <summary>
/// Subtle aim magnetism during slingshot pull. When trajectory prediction dots
/// pass near an unpopped balloon, gently nudges the aim toward that balloon
/// and gives it a subtle scale pulse as visual feedback.
///
/// Requires SlingshotInput on the same GameObject.
/// Reads balloon positions from BalloonNode instances in the scene.
/// Configurable via GameConfigSO.aimAssistEnabled.
/// </summary>
[RequireComponent(typeof(SlingshotInput))]
public class AimAssist : MonoBehaviour
{
    private SlingshotInput _input;
    private BalloonNode _highlightedBalloon;
    private Vector3 _highlightOriginalScale;
    private bool _isHighlighting;

    /// <summary>
    /// The aim adjustment vector applied this frame (read by SlingshotInput or visuals).
    /// </summary>
    public Vector2 AimAdjustment { get; private set; }

    /// <summary>
    /// The balloon currently being targeted by aim assist, if any.
    /// </summary>
    public BalloonNode TargetBalloon => _highlightedBalloon;

    private void Awake()
    {
        _input = GetComponent<SlingshotInput>();
    }

    private void LateUpdate()
    {
        if (!IsEnabled())
        {
            ClearHighlight();
            AimAdjustment = Vector2.zero;
            return;
        }

        if (!_input.IsAiming)
        {
            ClearHighlight();
            AimAdjustment = Vector2.zero;
            return;
        }

        Vector3 velocity = _input.PreviewVelocity;
        if (velocity.sqrMagnitude < 1f)
        {
            ClearHighlight();
            AimAdjustment = Vector2.zero;
            return;
        }

        FindNearestBalloonOnTrajectory(velocity);
        UpdateHighlightPulse();
    }

    private void OnDisable()
    {
        ClearHighlight();
        AimAdjustment = Vector2.zero;
    }

    private bool IsEnabled()
    {
        // Check GameConfigSO if available
        var config = FindGameConfig();
        if (config != null)
        {
            return config.aimAssistEnabled;
        }

        // Fallback: always enabled
        return true;
    }

    private float GetProximity()
    {
        var config = FindGameConfig();
        return config != null
            ? config.aimAssistProximity
            : GameConstants.AIM_ASSIST_PROXIMITY;
    }

    private float GetStrength()
    {
        var config = FindGameConfig();
        return config != null
            ? config.aimAssistStrength
            : GameConstants.AIM_ASSIST_STRENGTH;
    }

    private void FindNearestBalloonOnTrajectory(Vector3 velocity)
    {
        Vector3 origin = GameConstants.LAUNCH_POSITION;
        Vector3 gravity = Physics.gravity;
        float proximity = GetProximity();
        float proximitySqr = proximity * proximity;

        BalloonNode nearest = null;
        float nearestDistSqr = float.MaxValue;
        Vector3 nearestTrajectoryPoint = Vector3.zero;

        // Sample trajectory at same resolution as visual dots
        int sampleCount = GameConstants.TRAJECTORY_DOT_COUNT;
        float duration = GameConstants.TRAJECTORY_DURATION;

        // Get all unpopped balloons
        var balloons = FindObjectsByType<BalloonNode>(FindObjectsSortMode.None);

        for (int i = 0; i < sampleCount; i++)
        {
            float t = (float)i / (sampleCount - 1) * duration;
            Vector3 point = origin + velocity * t + 0.5f * gravity * t * t;

            for (int b = 0; b < balloons.Length; b++)
            {
                if (balloons[b].IsPopped)
                {
                    continue;
                }

                float distSqr = (balloons[b].transform.position - point).sqrMagnitude;
                if (distSqr < proximitySqr && distSqr < nearestDistSqr)
                {
                    nearestDistSqr = distSqr;
                    nearest = balloons[b];
                    nearestTrajectoryPoint = point;
                }
            }
        }

        // Update highlight target
        if (nearest != _highlightedBalloon)
        {
            ClearHighlight();

            if (nearest != null)
            {
                _highlightedBalloon = nearest;
                _highlightOriginalScale = nearest.transform.localScale;
                _isHighlighting = true;
            }
        }

        // Calculate aim adjustment
        if (nearest != null)
        {
            Vector3 balloonPos = nearest.transform.position;
            Vector2 offset = new Vector2(
                balloonPos.x - nearestTrajectoryPoint.x,
                balloonPos.y - nearestTrajectoryPoint.y);

            AimAdjustment = offset * GetStrength();
        }
        else
        {
            AimAdjustment = Vector2.zero;
        }
    }

    private void UpdateHighlightPulse()
    {
        if (!_isHighlighting || _highlightedBalloon == null)
        {
            return;
        }

        // Bail if the balloon was popped while highlighted
        if (_highlightedBalloon.IsPopped)
        {
            ClearHighlight();
            return;
        }

        float pulse = Mathf.Sin(Time.time * GameConstants.AIM_ASSIST_HIGHLIGHT_SPEED);
        float scaleMultiplier = Mathf.Lerp(1f, GameConstants.AIM_ASSIST_HIGHLIGHT_SCALE,
            (pulse + 1f) * 0.5f);

        _highlightedBalloon.transform.localScale = _highlightOriginalScale * scaleMultiplier;
    }

    private void ClearHighlight()
    {
        if (_isHighlighting && _highlightedBalloon != null && !_highlightedBalloon.IsPopped)
        {
            _highlightedBalloon.transform.localScale = _highlightOriginalScale;
        }

        _highlightedBalloon = null;
        _isHighlighting = false;
    }

    // Cache the config lookup to avoid repeated Resources.Load calls
    private static GameConfigSO _cachedConfig;
    private static bool _configSearched;

    private static GameConfigSO FindGameConfig()
    {
        if (_configSearched)
        {
            return _cachedConfig;
        }

        _configSearched = true;

        // Try the singleton pattern from Plan 01
        var configType = System.Type.GetType("GameConfigSO");
        if (configType != null)
        {
            var instanceProp = configType.GetProperty("Instance",
                System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.Static);
            if (instanceProp != null)
            {
                _cachedConfig = instanceProp.GetValue(null) as GameConfigSO;
                return _cachedConfig;
            }
        }

        // Fallback: try Resources
        _cachedConfig = Resources.Load<GameConfigSO>("GameConfig");
        return _cachedConfig;
    }
}
```
