```csharp
    public static readonly Color ProgressBarFillStart = new(0f, 0.9f, 0.3f, 1f);

    public static readonly Color ComboTier3 = new(1f, 0.6f, 0.1f, 1f);   // x5-x6: orange
    public static readonly Color ComboTier4 = new(1f, 0.15f, 0.15f, 1f); // x7+: red

    // --- Rarity Borders (used by perk/item cards in later 06.x plans) ---
    public static readonly Color CommonWhite = new(0.8f, 0.8f, 0.8f, 1f);
    public static readonly Color RareBlueBorder = new(0.2f, 0.5f, 1f, 1f);
    public static readonly Color EpicPurpleBorder = new(0.6f, 0.2f, 1f, 1f);
    public static readonly Color LegendaryGoldBorder = new(1f, 0.8f, 0f, 1f);

    // --- Shared UI ---
    public static readonly Color PanelBackground = new(0.06f, 0.06f, 0.08f, 0.85f);
    public static readonly Color ShadowBlack = new(0f, 0f, 0f, 0.8f);
    public static readonly Color ClearedGreen = new(0.3f, 0.9f, 0.3f, 1f);
    public static readonly Color FailedRed = new(0.9f, 0.3f, 0.3f, 1f);

    // --- Balloon Pop Text (matches BalloonColor enum order) ---

    /// <summary>
    /// Returns a neon-tinted UI color for floating score text based on balloon color.
    /// </summary>
    public static Color GetBalloonTextColor(BalloonColor balloonColor)
    {
        return balloonColor switch
        {
            BalloonColor.Red    => new Color(1f, 0.35f, 0.35f, 1f),
            BalloonColor.Blue   => new Color(0.35f, 0.6f, 1f, 1f),
            BalloonColor.Yellow => new Color(1f, 0.95f, 0.3f, 1f),
            BalloonColor.Green  => new Color(0.3f, 0.95f, 0.4f, 1f),
            BalloonColor.Purple => new Color(0.7f, 0.35f, 1f, 1f),
            _                   => ScoreWhite,
        };
    }

    /// <summary>
    /// Returns the combo color for a given combo count using the tier ramp.
    /// </summary>
    public static Color GetComboColor(int comboCount)
    {
        return comboCount switch
        {
            <= 2 => ComboTier1,
            <= 4 => ComboTier2,
            <= 6 => ComboTier3,
            _    => ComboTier4,
        };
    }
}
```

```csharp
// Assets/Scripts/BalloonGame/UI/InGameHUD.cs
using UnityEngine;
using UnityEngine.UI;
using TMPro;
using System.Collections;

/// <summary>
/// Premium neon noir in-game HUD. Replaces the old GameHUD entirely.
/// Builds its own Canvas hierarchy at runtime. Subscribes to EventBus for all updates.
/// Requires a UIConfigSO reference (assigned via Inspector or found via Resources).
/// </summary>
public class InGameHUD : MonoBehaviour
{
    [Header("Configuration")]
    [SerializeField] private UIConfigSO _uiConfig;

    // --- Canvas hierarchy ---
    private Canvas _canvas;
    private RectTransform _safeAreaPanel;

    // --- Top bar elements ---
    private TextMeshProUGUI _roomBadgeText;
    private TextMeshProUGUI _scoreText;
    private TextMeshProUGUI _dartsLabel;
    private Image _progressBarTrack;
    private Image _progressBarFill;
    private CanvasGroup _progressBarGlowGroup;
    private Image _progressBarGlow;

    // --- Dart icons ---
    private Image[] _dartIcons;
    private const int MaxDartIcons = 8;

    // --- Ink currency ---
    private TextMeshProUGUI _inkCountText;

    // --- Center message ---
    private TextMeshProUGUI _messageText;
    private CanvasGroup _messageGroup;

    // --- Score animation state ---
    private int _displayedScore;
    private int _actualScore;
    private int _targetScore;
    private float _progressBarTarget;
    private float _progressBarCurrent;

    // --- Room state ---
    private int _currentRoom = 1;
    private int _dartsRemaining;
    private int _inkCount;

    #region Lifecycle

    private void Awake()
    {
        if (_uiConfig == null)
        {
            _uiConfig = Resources.Load<UIConfigSO>("UIConfig");
        }

        BuildCanvas();
    }

    private void OnEnable()
    {
        EventBus.Subscribe<ScoreChangedEvent>(OnScoreChanged);
        EventBus.Subscribe<RoomStateChangedEvent>(OnRoomStateChanged);
        EventBus.Subscribe<DartFinishedEvent>(OnDartFinished);
        EventBus.Subscribe<DartLaunchedEvent>(OnDartLaunched);
    }

    private void OnDisable()
    {
        EventBus.Unsubscribe<ScoreChangedEvent>(OnScoreChanged);
        EventBus.Unsubscribe<RoomStateChangedEvent>(OnRoomStateChanged);
        EventBus.Unsubscribe<DartFinishedEvent>(OnDartFinished);
        EventBus.Unsubscribe<DartLaunchedEvent>(OnDartLaunched);
    }

    private void Update()
    {
        AnimateScoreCounter();
        AnimateProgressBar();
        AnimateProgressBarGlow();
    }

    #endregion

    #region Canvas Construction

    private void BuildCanvas()
    {
        // --- Root Canvas ---
        var canvasGo = new GameObject("InGameHUDCanvas");
        canvasGo.transform.SetParent(transform, false);

        _canvas = canvasGo.AddComponent<Canvas>();
        _canvas.renderMode = RenderMode.ScreenSpaceOverlay;
        _canvas.sortingOrder = 100;

        var scaler = canvasGo.AddComponent<CanvasScaler>();
        scaler.uiScaleMode = CanvasScaler.ScaleMode.ScaleWithScreenSize;
        scaler.referenceResolution = new Vector2(1080f, 1920f);
        scaler.matchWidthOrHeight = 0.5f;

        canvasGo.AddComponent<GraphicRaycaster>();

        // --- Safe Area Panel ---
        var safeAreaGo = new GameObject("SafeAreaPanel");
        safeAreaGo.transform.SetParent(canvasGo.transform, false);
        _safeAreaPanel = safeAreaGo.AddComponent<RectTransform>();
        _safeAreaPanel.anchorMin = Vector2.zero;
        _safeAreaPanel.anchorMax = Vector2.one;
        _safeAreaPanel.offsetMin = Vector2.zero;
        _safeAreaPanel.offsetMax = Vector2.zero;
        safeAreaGo.AddComponent<SafeAreaHandler>();

        // --- Build sections ---
        BuildTopBar();
        BuildDartIcons();
        BuildInkDisplay();
        BuildMessageDisplay();
    }

    private void BuildTopBar()
    {
        // --- Top bar container ---
        var topBarGo = CreateUIObject("TopBar", _safeAreaPanel);
        var topBarRect = topBarGo.GetComponent<RectTransform>();
        topBarRect.anchorMin = new Vector2(0f, 1f);
        topBarRect.anchorMax = new Vector2(1f, 1f);
        topBarRect.pivot = new Vector2(0.5f, 1f);
        topBarRect.anchoredPosition = new Vector2(0f, -10f);
        topBarRect.sizeDelta = new Vector2(0f, 120f);

        // --- Room Badge (top left) ---
        BuildRoomBadge(topBarRect);

        // --- Score + Progress Bar (top center) ---
        BuildScoreSection(topBarRect);
    }

    private void BuildRoomBadge(RectTransform parent)
    {
        var badgeGo = CreateUIObject("RoomBadge", parent);
        var badgeRect = badgeGo.GetComponent<RectTransform>();
        badgeRect.anchorMin = new Vector2(0f, 1f);
        badgeRect.anchorMax = new Vector2(0f, 1f);
        badgeRect.pivot = new Vector2(0f, 1f);
        badgeRect.anchoredPosition = new Vector2(24f, -8f);
        badgeRect.sizeDelta = new Vector2(180f, 50f);

        // Badge background
        var badgeBg = badgeGo.AddComponent<Image>();
        badgeBg.color = UIColors.PanelBackground;

        // Badge text
        _roomBadgeText = CreateTMPText(badgeRect, "RoomText", "ROOM 1",
            28f, UIColors.RoomPink, TextAlignmentOptions.Center);
        var roomTextRect = _roomBadgeText.rectTransform;
        roomTextRect.anchorMin = Vector2.zero;
        roomTextRect.anchorMax = Vector2.one;
        roomTextRect.offsetMin = Vector2.zero;
        roomTextRect.offsetMax = Vector2.zero;
    }

    private void BuildScoreSection(RectTransform parent)
    {
        // --- Score container (center) ---
        var scoreContainerGo = CreateUIObject("ScoreContainer", parent);
        var scoreContainerRect = scoreContainerGo.GetComponent<RectTransform>();
        scoreContainerRect.anchorMin = new Vector2(0.5f, 1f);
        scoreContainerRect.anchorMax = new Vector2(0.5f, 1f);
        scoreContainerRect.pivot = new Vector2(0.5f, 1f);
        scoreContainerRect.anchoredPosition = new Vector2(0f, -8f);
        scoreContainerRect.sizeDelta = new Vector2(400f, 90f);

        // Score number text
        _scoreText = CreateTMPText(scoreContainerRect, "ScoreValue", "0",
            48f, UIColors.ScoreWhite, TextAlignmentOptions.Center);
        var scoreValueRect = _scoreText.rectTransform;
        scoreValueRect.anchorMin = new Vector2(0f, 0.45f);
        scoreValueRect.anchorMax = new Vector2(1f, 1f);
        scoreValueRect.offsetMin = Vector2.zero;
        scoreValueRect.offsetMax = Vector2.zero;

        // --- Progress bar track ---
        var trackGo = CreateUIObject("ProgressTrack", scoreContainerRect);
        var trackRect = trackGo.GetComponent<RectTransform>();
        trackRect.anchorMin = new Vector2(0.05f, 0.1f);
        trackRect.anchorMax = new Vector2(0.95f, 0.35f);
        trackRect.offsetMin = Vector2.zero;
        trackRect.offsetMax = Vector2.zero;
        _progressBarTrack = trackGo.AddComponent<Image>();
        _progressBarTrack.color = UIColors.ProgressBarTrack;

        // --- Progress bar fill ---
        var fillGo = CreateUIObject("ProgressFill", trackRect);
        var fillRect = fillGo.GetComponent<RectTransform>();
        fillRect.anchorMin = Vector2.zero;
        fillRect.anchorMax = Vector2.one;
        fillRect.offsetMin = Vector2.zero;
        fillRect.offsetMax = Vector2.zero;
        _progressBarFill = fillGo.AddComponent<Image>();
        _progressBarFill.color = UIColors.ProgressBarFillStart;
        _progressBarFill.type = Image.Type.Filled;
        _progressBarFill.fillMethod = Image.FillMethod.Horizontal;
        _progressBarFill.fillAmount = 0f;

        // --- Progress bar glow overlay ---
        var glowGo = CreateUIObject("ProgressGlow", trackRect);
        var glowRect = glowGo.GetComponent<RectTransform>();
        glowRect.anchorMin = new Vector2(0f, -0.5f);
        glowRect.anchorMax = new Vector2(1f, 1.5f);
        glowRect.offsetMin = new Vector2(-4f, 0f);
        glowRect.offsetMax = new Vector2(4f, 0f);
        _progressBarGlowGroup = glowGo.AddComponent<CanvasGroup>();
        _progressBarGlowGroup.alpha = 0f;
        _progressBarGlow = glowGo.AddComponent<Image>();
        _progressBarGlow.color = UIColors.ProgressBarGlow;
        _progressBarGlow.raycastTarget = false;
    }

    private void BuildDartIcons()
    {
        var dartsContainerGo = CreateUIObject("DartsContainer", _safeAreaPanel);
        var dartsContainerRect = dartsContainerGo.GetComponent<RectTransform>();
        dartsContainerRect.anchorMin = new Vector2(1f, 1f);
        dartsContainerRect.anchorMax = new Vector2(1f, 1f);
        dartsContainerRect.pivot = new Vector2(1f, 1f);
        dartsContainerRect.anchoredPosition = new Vector2(-24f, -18f);
        dartsContainerRect.sizeDelta = new Vector2(200f, 40f);

        var hlg = dartsContainerGo.AddComponent<HorizontalLayoutGroup>();
        hlg.spacing = 6f;
        hlg.childAlignment = TextAnchor.MiddleRight;
        hlg.childControlWidth = false;
        hlg.childControlHeight = false;
        hlg.childForceExpandWidth = false;
        hlg.childForceExpandHeight = false;

        _dartIcons = new Image[MaxDartIcons];
        for (int i = 0; i < MaxDartIcons; i++)
        {
            var iconGo = CreateUIObject($"DartIcon_{i}", dartsContainerRect);
            var iconRect = iconGo.GetComponent<RectTransform>();
            iconRect.sizeDelta = new Vector2(20f, 20f);

            var iconImg = iconGo.AddComponent<Image>();
            iconImg.color = UIColors.DartBlue;
            // Diamond shape via default sprite; filled = available, empty = used
            _dartIcons[i] = iconImg;
            iconGo.SetActive(false);
        }
    }

    private void BuildInkDisplay()
    {
        var inkGo = CreateUIObject("InkDisplay", _safeAreaPanel);
        var inkRect = inkGo.GetComponent<RectTransform>();
        inkRect.anchorMin = new Vector2(1f, 1f);
        inkRect.anchorMax = new Vector2(1f, 1f);
        inkRect.pivot = new Vector2(1f, 1f);
        inkRect.anchoredPosition = new Vector2(-24f, -62f);
        inkRect.sizeDelta = new Vector2(120f, 30f);

        _inkCountText = CreateTMPText(inkRect, "InkCount", "0",
            24f, UIColors.InkCyan, TextAlignmentOptions.Right);
        var inkTextRect = _inkCountText.rectTransform;
        inkTextRect.anchorMin = Vector2.zero;
        inkTextRect.anchorMax = Vector2.one;
        inkTextRect.offsetMin = Vector2.zero;
        inkTextRect.offsetMax = Vector2.zero;

    {
        var messageGo = CreateUIObject("MessageDisplay", _safeAreaPanel);
        var messageRect = messageGo.GetComponent<RectTransform>();
        messageRect.anchorMin = new Vector2(0.5f, 0.5f);
        messageRect.anchorMax = new Vector2(0.5f, 0.5f);
        messageRect.pivot = new Vector2(0.5f, 0.5f);
        messageRect.anchoredPosition = Vector2.zero;
        messageRect.sizeDelta = new Vector2(800f, 200f);

        _messageGroup = messageGo.AddComponent<CanvasGroup>();
        _messageGroup.alpha = 0f;

        float fontSize = _uiConfig != null ? _uiConfig.messageFontSize : 72f;
        _messageText = CreateTMPText(messageRect, "MessageText", "",
            fontSize, UIColors.ScoreWhite, TextAlignmentOptions.Center);
        var msgTextRect = _messageText.rectTransform;
        msgTextRect.anchorMin = Vector2.zero;
        msgTextRect.anchorMax = Vector2.one;
        msgTextRect.offsetMin = Vector2.zero;
        msgTextRect.offsetMax = Vector2.zero;

        messageGo.SetActive(false);

    #endregion

    #region UI Factory Helpers

    private static GameObject CreateUIObject(string name, RectTransform parent)
    {
        var go = new GameObject(name);
        go.transform.SetParent(parent, false);
        go.AddComponent<RectTransform>();
        return go;
    }

    private static TextMeshProUGUI CreateTMPText(
        RectTransform parent, string name, string initialText,
        float fontSize, Color color, TextAlignmentOptions alignment)
    {
        var go = new GameObject(name);
        go.transform.SetParent(parent, false);
        go.AddComponent<RectTransform>();

        var tmp = go.AddComponent<TextMeshProUGUI>();
        tmp.text = initialText;
        tmp.fontSize = fontSize;
        tmp.color = color;
        tmp.alignment = alignment;
        tmp.enableWordWrapping = false;
        tmp.overflowMode = TextOverflowModes.Overflow;
        tmp.raycastTarget = false;

        // Drop shadow via TMP material — subtle black offset
        tmp.fontSharedMaterial.EnableKeyword("UNDERLAY_ON");
        tmp.fontSharedMaterial.SetColor("_UnderlayColor", UIColors.ShadowBlack);
        tmp.fontSharedMaterial.SetFloat("_UnderlayOffsetX", 1f);
        tmp.fontSharedMaterial.SetFloat("_UnderlayOffsetY", -1f);
        tmp.fontSharedMaterial.SetFloat("_UnderlaySoftness", 0.2f);

        return tmp;
    }

    #endregion

    #region EventBus Handlers

    private void OnScoreChanged(ScoreChangedEvent e)
    {
        _actualScore = e.CurrentScore;
        _targetScore = e.TargetScore;
        _progressBarTarget = _targetScore > 0
            ? Mathf.Clamp01((float)_actualScore / _targetScore)
            : 0f;
    }

    private void OnRoomStateChanged(RoomStateChangedEvent e)
    {
        _dartsRemaining = e.DartsRemaining;
        RefreshDartIcons();

        switch (e.NewState)
        {
            case BalloonGameManager.GameState.RoomIntro:
                _currentRoom++;
                _roomBadgeText.text = $"ROOM {_currentRoom}";
                _displayedScore = 0;
                _actualScore = 0;
                _progressBarCurrent = 0f;
                _progressBarTarget = 0f;
                _progressBarFill.fillAmount = 0f;
                _scoreText.text = "0";
                HideMessage();
                break;

            case BalloonGameManager.GameState.RoomCleared:
                ShowMessage("CLEARED!", UIColors.ClearedGreen);
                break;

            case BalloonGameManager.GameState.RoomFailed:
                ShowMessage("FAILED", UIColors.FailedRed);
                break;
        }
    }

    private void OnDartLaunched(DartLaunchedEvent e)
    {
        _dartsRemaining--;
        RefreshDartIcons();
    }

    private void OnDartFinished(DartFinishedEvent e)
    {
        // Dart icons already updated on launch; no action needed here
    }

    #endregion

    #region Animation

    private void AnimateScoreCounter()
    {
        if (_displayedScore == _actualScore) return;

        float speed = _uiConfig != null ? _uiConfig.scoreCountSpeed : 2000f;
        float step = speed * Time.deltaTime;

        if (_displayedScore < _actualScore)
        {
            _displayedScore = Mathf.Min(_actualScore, _displayedScore + Mathf.CeilToInt(step));
        }
        else
        {
            _displayedScore = Mathf.Max(_actualScore, _displayedScore - Mathf.CeilToInt(step));
        }

        _scoreText.text = _displayedScore.ToString();
    }

    private void AnimateProgressBar()
    {
        float speed = _uiConfig != null ? _uiConfig.progressBarLerpSpeed : 4f;

        _progressBarCurrent = Mathf.MoveTowards(
            _progressBarCurrent, _progressBarTarget, speed * Time.deltaTime);
        _progressBarFill.fillAmount = _progressBarCurrent;

        // Color lerp: green at 0% → yellow at 100%
        _progressBarFill.color = Color.Lerp(
            UIColors.ProgressBarFillStart, UIColors.ProgressBarFillEnd, _progressBarCurrent);
    }

    private void AnimateProgressBarGlow()
    {
        float threshold = _uiConfig != null ? _uiConfig.progressBarGlowThreshold : 0.75f;
        float pulseSpeed = _uiConfig != null ? _uiConfig.progressBarGlowPulseSpeed : 2f;

        if (_progressBarCurrent >= threshold)
        {
            float pulse = (Mathf.Sin(Time.time * pulseSpeed * Mathf.PI * 2f) + 1f) * 0.5f;
            _progressBarGlowGroup.alpha = Mathf.Lerp(0.2f, 0.6f, pulse);
        }
        else
        {
            _progressBarGlowGroup.alpha = 0f;
        }
    }

    #endregion

    #region Dart Icons

    private void RefreshDartIcons()
    {
        int total = GameConstants.STARTING_DARTS;
        for (int i = 0; i < MaxDartIcons; i++)
        {
            if (i < total)
            {
                _dartIcons[i].gameObject.SetActive(true);
                bool available = i < _dartsRemaining;
                _dartIcons[i].color = available
                    ? UIColors.DartBlue
                    : new Color(UIColors.DartBlue.r, UIColors.DartBlue.g, UIColors.DartBlue.b, 0.25f);
            }
            else
            {
                _dartIcons[i].gameObject.SetActive(false);
            }
        }
    }

    #endregion

    #region Message Display

    private void ShowMessage(string text, Color color)
    {
        _messageText.text = text;
        _messageText.color = color;
        _messageText.gameObject.transform.parent.gameObject.SetActive(true);
        StartCoroutine(AnimateMessageAppear());
    }

    private void HideMessage()
    {
```

```csharp
// Assets/Scripts/BalloonGame/UI/ComboDisplay.cs
using UnityEngine;
using TMPro;
using System.Collections;

/// <summary>
/// Displays the combo multiplier ("×N") at bottom-center of the screen.
/// Only visible when combo count >= 2. Punch-scales on each increment.
/// Color ramps through tiers. Fades out when combo resets.
/// Attaches to the same Canvas as InGameHUD (finds it automatically).
/// </summary>
public class ComboDisplay : MonoBehaviour
{
    [SerializeField] private UIConfigSO _uiConfig;

    private TextMeshProUGUI _comboText;
    private CanvasGroup _canvasGroup;
    private RectTransform _rectTransform;

    private int _lastComboCount;
    private Coroutine _punchCoroutine;
    private Coroutine _fadeCoroutine;

    private void Awake()
    {
        if (_uiConfig == null)
        {
            _uiConfig = Resources.Load<UIConfigSO>("UIConfig");
        }

        BuildDisplay();
    }

    private void OnEnable()
    {
        EventBus.Subscribe<ComboChangedEvent>(OnComboChanged);
    }

    private void OnDisable()
    {
        EventBus.Unsubscribe<ComboChangedEvent>(OnComboChanged);
    }

    private void BuildDisplay()
    {
        // Find the InGameHUD canvas in the scene
        Canvas hudCanvas = null;
        var canvases = FindObjectsByType<Canvas>(FindObjectsSortMode.None);
        foreach (var c in canvases)
        {
            if (c.gameObject.name == "InGameHUDCanvas")
            {
                hudCanvas = c;
                break;
            }
        }

        // Fallback: create on own transform if canvas not found
        Transform parent = hudCanvas != null ? hudCanvas.transform : transform;

        var comboGo = new GameObject("ComboDisplay");
        comboGo.transform.SetParent(parent, false);

        _rectTransform = comboGo.AddComponent<RectTransform>();
        _rectTransform.anchorMin = new Vector2(0.5f, 0f);
        _rectTransform.anchorMax = new Vector2(0.5f, 0f);
        _rectTransform.pivot = new Vector2(0.5f, 0f);
        _rectTransform.anchoredPosition = new Vector2(0f, 280f);
        _rectTransform.sizeDelta = new Vector2(300f, 120f);

        _canvasGroup = comboGo.AddComponent<CanvasGroup>();
        _canvasGroup.alpha = 0f;

        var textGo = new GameObject("ComboText");
        textGo.transform.SetParent(comboGo.transform, false);
        var textRect = textGo.AddComponent<RectTransform>();
        textRect.anchorMin = Vector2.zero;
        textRect.anchorMax = Vector2.one;
        textRect.offsetMin = Vector2.zero;
        textRect.offsetMax = Vector2.zero;

        _comboText = textGo.AddComponent<TextMeshProUGUI>();
        _comboText.text = "";
        _comboText.fontSize = 64f;
        _comboText.color = UIColors.ComboGold;
        _comboText.alignment = TextAlignmentOptions.Center;
        _comboText.enableWordWrapping = false;
        _comboText.raycastTarget = false;
        _comboText.fontStyle = FontStyles.Bold;
    }

    private void OnComboChanged(ComboChangedEvent e)
    {
        if (e.WasReset)
        {
            // Combo ended — fade out
            if (_punchCoroutine != null) StopCoroutine(_punchCoroutine);
            if (_fadeCoroutine != null) StopCoroutine(_fadeCoroutine);
            _fadeCoroutine = StartCoroutine(FadeOut());
            _lastComboCount = 0;
            return;
        }

        if (e.ComboCount < 2)
        {
            // Not enough for display
            return;
        }

        // Update text and color
        _comboText.text = $"\u00d7{e.ComboCount}";
        _comboText.color = UIColors.GetComboColor(e.ComboCount);

        // Show if hidden
        if (_fadeCoroutine != null) StopCoroutine(_fadeCoroutine);
        _canvasGroup.alpha = 1f;

        // Punch scale animation
        if (_punchCoroutine != null) StopCoroutine(_punchCoroutine);
        _punchCoroutine = StartCoroutine(PunchScale());

        _lastComboCount = e.ComboCount;
    }

    private IEnumerator PunchScale()
    {
        float duration = _uiConfig != null ? _uiConfig.comboPunchDuration : 0.25f;
        float peakScale = _uiConfig != null ? _uiConfig.comboPunchScale : 1.4f;
        float elapsed = 0f;

        while (elapsed < duration)
        {
            elapsed += Time.deltaTime;
            float t = elapsed / duration;

            // Quick overshoot then settle: sin curve from 0 to PI
            float scaleFactor = 1f + (peakScale - 1f) * Mathf.Sin(t * Mathf.PI);
            _rectTransform.localScale = new Vector3(scaleFactor, scaleFactor, 1f);

            yield return null;
        }

        _rectTransform.localScale = Vector3.one;
        _punchCoroutine = null;
    }

    private IEnumerator FadeOut()
    {
        float duration = _uiConfig != null ? _uiConfig.comboFadeDuration : 0.5f;
        float startAlpha = _canvasGroup.alpha;
        float elapsed = 0f;

        while (elapsed < duration)
        {
            elapsed += Time.deltaTime;
            float t = elapsed / duration;
            _canvasGroup.alpha = Mathf.Lerp(startAlpha, 0f, t);
            yield return null;
        }

        _canvasGroup.alpha = 0f;
        _fadeCoroutine = null;
    }
}
```

```csharp
// Assets/Scripts/BalloonGame/UI/FloatingScoreText.cs
using UnityEngine;
using TMPro;
using System.Collections;

/// <summary>
/// Manages a pool of world-space TextMeshPro objects that display "+100" style
/// floating score text at balloon pop positions. Rises upward and fades out.
/// Subscribes to BalloonPoppedEvent via EventBus.
/// </summary>
public class FloatingScoreText : MonoBehaviour
{
    [SerializeField] private UIConfigSO _uiConfig;

    private TextMeshPro[] _pool;
    private int _poolIndex;
    private Camera _mainCamera;

    private void Awake()
    {
        if (_uiConfig == null)
        {
            _uiConfig = Resources.Load<UIConfigSO>("UIConfig");
        }

        _mainCamera = Camera.main;
        BuildPool();
    }

    private void OnEnable()
    {
        EventBus.Subscribe<BalloonPoppedEvent>(OnBalloonPopped);
    }

    private void OnDisable()
    {
        EventBus.Unsubscribe<BalloonPoppedEvent>(OnBalloonPopped);
    }

    private void BuildPool()
    {
        int poolSize = _uiConfig != null ? _uiConfig.floatingTextPoolSize : 10;
        _pool = new TextMeshPro[poolSize];

        for (int i = 0; i < poolSize; i++)
        {
            var go = new GameObject($"FloatingText_{i}");
            go.transform.SetParent(transform, false);

            var tmp = go.AddComponent<TextMeshPro>();
            tmp.fontSize = _uiConfig != null ? _uiConfig.floatingTextFontSize : 8f;
            tmp.alignment = TextAlignmentOptions.Center;
            tmp.enableWordWrapping = false;
            tmp.sortingOrder = 200;
            tmp.raycastTarget = false;
            tmp.fontStyle = FontStyles.Bold;

            go.SetActive(false);
            _pool[i] = tmp;
        }
    }

    private void OnBalloonPopped(BalloonPoppedEvent e)
    {
        SpawnText(e.WorldPosition, e.FinalPoints, e.Color);
    }

    /// <summary>
    /// Activates a pooled text object at the given world position.
    /// </summary>
    private void SpawnText(Vector3 worldPosition, int points, BalloonColor balloonColor)
    {
        TextMeshPro tmp = GetNextFromPool();
        if (tmp == null) return;

        tmp.text = $"+{points}";
        tmp.color = UIColors.GetBalloonTextColor(balloonColor);

        // Position in world space, slightly in front of balloons
        tmp.transform.position = new Vector3(worldPosition.x, worldPosition.y, worldPosition.z - 0.5f);

        // Face the camera
        if (_mainCamera != null)
        {
            tmp.transform.rotation = _mainCamera.transform.rotation;
        }

        tmp.gameObject.SetActive(true);
        StartCoroutine(AnimateFloatAndFade(tmp));
    }

    private TextMeshPro GetNextFromPool()
    {
        if (_pool == null || _pool.Length == 0) return null;

        // Round-robin through pool
        TextMeshPro tmp = _pool[_poolIndex];
        _poolIndex = (_poolIndex + 1) % _pool.Length;

        // If currently animating, force-deactivate for reuse
        tmp.gameObject.SetActive(false);
        return tmp;
    }

    private IEnumerator AnimateFloatAndFade(TextMeshPro tmp)
    {
        float duration = _uiConfig != null ? _uiConfig.floatingTextDuration : 0.8f;
        float riseDistance = _uiConfig != null ? _uiConfig.floatingTextRisePx : 1.5f;

        Vector3 startPos = tmp.transform.position;
        Vector3 endPos = startPos + new Vector3(0f, riseDistance, 0f);
        Color startColor = tmp.color;
        float elapsed = 0f;

        while (elapsed < duration)
        {
            elapsed += Time.deltaTime;
            float t = elapsed / duration;

            // Ease-out position
            float easedT = 1f - (1f - t) * (1f - t);
            tmp.transform.position = Vector3.Lerp(startPos, endPos, easedT);

            // Linear alpha fade
            Color c = startColor;
            c.a = Mathf.Lerp(1f, 0f, t);
            tmp.color = c;

            yield return null;
        }

        tmp.gameObject.SetActive(false);
    }
}
```
