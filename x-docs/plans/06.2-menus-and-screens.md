### `Assets/Scripts/BalloonGame/UI/Screens/ScreenTransition.cs`

```csharp
using System;
using System.Collections;
using UnityEngine;
using UnityEngine.UI;

/// <summary>
/// Lazy singleton for full-screen fade transitions.
/// Creates its own Canvas (sortingOrder 999) on first access.
/// Uses coroutines — no DOTween dependency.
/// </summary>
public class ScreenTransition : MonoBehaviour
{
    private static ScreenTransition _instance;
    private CanvasGroup _canvasGroup;
    private Coroutine _activeTransition;

    private static ScreenTransition Instance
    {
        get
        {
            if (_instance == null)
            {
                CreateInstance();
            }
            return _instance;
        }
    }

    private static void CreateInstance()
    {
        var go = new GameObject("[ScreenTransition]");
        DontDestroyOnLoad(go);

        // Canvas
        var canvas = go.AddComponent<Canvas>();
        canvas.renderMode = RenderMode.ScreenSpaceOverlay;
        canvas.sortingOrder = 999;

        // CanvasScaler
        var scaler = go.AddComponent<CanvasScaler>();
        scaler.uiScaleMode = CanvasScaler.ScaleMode.ScaleWithScreenSize;
        scaler.referenceResolution = new Vector2(1080, 1920);
        scaler.matchWidthOrHeight = 0.5f;

        // Full-screen black Image
        var imageGo = new GameObject("BlackOverlay");
        imageGo.transform.SetParent(go.transform, false);

        var rect = imageGo.AddComponent<RectTransform>();
        rect.anchorMin = Vector2.zero;
        rect.anchorMax = Vector2.one;
        rect.offsetMin = Vector2.zero;
        rect.offsetMax = Vector2.zero;

        var image = imageGo.AddComponent<Image>();
        image.color = Color.black;
        image.raycastTarget = true;

        // CanvasGroup on the image for alpha control
        var canvasGroup = imageGo.AddComponent<CanvasGroup>();
        canvasGroup.alpha = 0f;
        canvasGroup.blocksRaycasts = false;
        canvasGroup.interactable = false;

        _instance = go.AddComponent<ScreenTransition>();
        _instance._canvasGroup = canvasGroup;
    }

    /// <summary>
    /// Fade to black, execute action while black, fade back.
    /// </summary>
    /// <param name="onBlack">Action to invoke while screen is fully black.</param>
    /// <param name="fadeIn">Duration to fade to black (seconds).</param>
    /// <param name="fadeOut">Duration to fade from black (seconds).</param>
    public static void FadeTo(Action onBlack, float fadeIn = 0.3f, float fadeOut = 0.3f)
    {
        Instance.StartFadeTo(onBlack, fadeIn, fadeOut);
    }

    /// <summary>
    /// Fade from current alpha to fully black.
    /// </summary>
    public static Coroutine FadeOut(float duration = 0.3f)
    {
        return Instance.StartFade(Instance._canvasGroup.alpha, 1f, duration);
    }

    /// <summary>
    /// Fade from current alpha to fully transparent.
    /// </summary>
    public static Coroutine FadeIn(float duration = 0.3f)
    {
        return Instance.StartFade(Instance._canvasGroup.alpha, 0f, duration);
    }

    private void StartFadeTo(Action onBlack, float fadeIn, float fadeOut)
    {
        if (_activeTransition != null)
        {
            StopCoroutine(_activeTransition);
        }
        _activeTransition = StartCoroutine(FadeToSequence(onBlack, fadeIn, fadeOut));
    }

    private Coroutine StartFade(float from, float to, float duration)
    {
        if (_activeTransition != null)
        {
            StopCoroutine(_activeTransition);
        }
        _activeTransition = StartCoroutine(LerpAlpha(from, to, duration));
        return _activeTransition;
    }

    private IEnumerator FadeToSequence(Action onBlack, float fadeIn, float fadeOut)
    {
        // Fade to black
        yield return LerpAlpha(_canvasGroup.alpha, 1f, fadeIn);

        // Execute callback while black
        onBlack?.Invoke();

        // Hold one frame so any scene changes settle
        yield return null;

        // Fade back to transparent
        yield return LerpAlpha(1f, 0f, fadeOut);

        _activeTransition = null;
    }

    private IEnumerator LerpAlpha(float from, float to, float duration)
    {
        _canvasGroup.blocksRaycasts = true;
        _canvasGroup.alpha = from;
        float elapsed = 0f;

        while (elapsed < duration)
        {
            elapsed += Time.unscaledDeltaTime;
            _canvasGroup.alpha = Mathf.Lerp(from, to, Mathf.Clamp01(elapsed / duration));
            yield return null;
        }

        _canvasGroup.alpha = to;
        _canvasGroup.blocksRaycasts = to > 0.01f;
        _activeTransition = null;
    }

    private void OnDestroy()
    {
        if (_instance == this)
        {
            _instance = null;
        }
    }
}
```

### `Assets/Scripts/BalloonGame/UI/Screens/TitleScreen.cs`

```csharp
using System.Collections;
using UnityEngine;
using UnityEngine.UI;
using TMPro;

/// <summary>
/// Title screen controller. Shows game title, tap-to-start, high score, and settings button.
/// Includes ambient mist particle system for atmosphere.
/// </summary>
public class TitleScreen : MonoBehaviour
{
    [Header("Text Elements")]
    [SerializeField] private TMP_Text _titleText;
    [SerializeField] private TMP_Text _subtitleText;
    [SerializeField] private TMP_Text _tapToStartText;
    [SerializeField] private TMP_Text _highScoreText;

    [Header("Buttons")]
    [SerializeField] private Button _settingsButton;

    [Header("References")]
    [SerializeField] private SettingsPanel _settingsPanel;
    [SerializeField] private CanvasGroup _screenGroup;

    [Header("Particles")]
    [SerializeField] private ParticleSystem _mistParticles;

    private RunManager _runManager;
    private bool _isTransitioning;
    private Coroutine _pulseCoroutine;

    private void Awake()
    {
        _runManager = FindAnyObjectByType<RunManager>();
    }

    private void OnEnable()
    {
        RefreshHighScore();
        SetupText();
        _pulseCoroutine = StartCoroutine(PulseTapToStart());

        if (_settingsButton != null)
        {
            _settingsButton.onClick.AddListener(OnSettingsPressed);
        }

        if (_mistParticles != null && !_mistParticles.isPlaying)
        {
            _mistParticles.Play();
        }

        _isTransitioning = false;
    }

    private void OnDisable()
    {
        if (_pulseCoroutine != null)
        {
            StopCoroutine(_pulseCoroutine);
            _pulseCoroutine = null;
        }

        if (_settingsButton != null)
        {
            _settingsButton.onClick.RemoveListener(OnSettingsPressed);
        }
    }

    private void Update()
    {
        if (_isTransitioning) return;

        // Detect tap / click anywhere (except when settings panel is open)
        if (_settingsPanel != null && _settingsPanel.IsOpen) return;

        if (Input.touchCount > 0 && Input.GetTouch(0).phase == TouchPhase.Began)
        {
            StartGame();
        }
        else if (Input.GetMouseButtonDown(0))
        {
            StartGame();
        }
    }

    private void SetupText()
    {
        if (_titleText != null)
        {
            _titleText.text = "INKSHOT";
            _titleText.fontSize = 120;
            _titleText.fontStyle = FontStyles.Bold;
            _titleText.alignment = TextAlignmentOptions.Center;
            _titleText.color = UIColors.NeonCyan;
        }

        if (_subtitleText != null)
        {
            _subtitleText.text = "A Neon Carnival Roguelite";
            _subtitleText.fontSize = 32;
            _subtitleText.fontStyle = FontStyles.Italic;
            _subtitleText.alignment = TextAlignmentOptions.Center;
            _subtitleText.color = UIColors.TextSecondary;
        }

        if (_tapToStartText != null)
        {
            _tapToStartText.text = "TAP TO START";
            _tapToStartText.fontSize = 42;
            _tapToStartText.fontStyle = FontStyles.Bold;
            _tapToStartText.alignment = TextAlignmentOptions.Center;
            _tapToStartText.color = UIColors.TextPrimary;
        }
    }

    private void RefreshHighScore()
    {
        if (_highScoreText == null) return;

        int highScore = 0;
        if (SaveManager.Instance != null)
        {
            highScore = SaveManager.Instance.Data.highScore;
        }

        if (highScore > 0)
        {
            _highScoreText.text = $"HIGH SCORE: {highScore:N0}";
            _highScoreText.fontSize = 28;
            _highScoreText.alignment = TextAlignmentOptions.Center;
            _highScoreText.color = UIColors.NeonGold;
            _highScoreText.gameObject.SetActive(true);
        }
        else
        {
            _highScoreText.gameObject.SetActive(false);
        }
    }

    private IEnumerator PulseTapToStart()
    {
        while (true)
        {
            if (_tapToStartText == null) yield break;

            float t = (Mathf.Sin(Time.unscaledTime * GameConstants.TITLE_PULSE_SPEED) + 1f) * 0.5f;
            float alpha = Mathf.Lerp(
                GameConstants.TITLE_PULSE_MIN_ALPHA,
                GameConstants.TITLE_PULSE_MAX_ALPHA,
                t
            );

            Color c = _tapToStartText.color;
            c.a = alpha;
            _tapToStartText.color = c;

            yield return null;
        }
    }

    private void StartGame()
    {
        _isTransitioning = true;

        ScreenTransition.FadeTo(() =>
        {
            gameObject.SetActive(false);

            if (_runManager != null)
            {
                _runManager.StartNewRun();
            }
            else
            {
                Debug.LogWarning("[TitleScreen] No RunManager found. Cannot start run.");
            }
        });
    }

    private void OnSettingsPressed()
    {
        if (_settingsPanel != null)
        {
            _settingsPanel.Open();
        }
    }

    /// <summary>
    /// Show the title screen (called when returning from a run).
    /// </summary>
    public void Show()
    {
        gameObject.SetActive(true);
    }

    /// <summary>
    /// Creates and configures the ambient mist ParticleSystem at runtime.
    /// Call once during scene setup if no prefab-based particle system is assigned.
    /// </summary>
    public static ParticleSystem CreateMistParticles(Transform parent)
    {
        var go = new GameObject("TitleMist");
        go.transform.SetParent(parent, false);
        go.transform.localPosition = Vector3.zero;

        var ps = go.AddComponent<ParticleSystem>();
        var main = ps.main;
        main.loop = true;
        main.startLifetime = 6f;
        main.startSpeed = 0.3f;
        main.startSize = new ParticleSystem.MinMaxCurve(1.5f, 3f);
        main.startColor = new Color(0.5f, 0.5f, 0.6f, 0.06f);
        main.maxParticles = GameConstants.TITLE_MIST_PARTICLE_COUNT;
        main.simulationSpace = ParticleSystemSimulationSpace.World;
        main.gravityModifier = 0f;

        var emission = ps.emission;
        emission.rateOverTime = 2f;

        var shape = ps.shape;
        shape.shapeType = ParticleSystemShapeType.Box;
        shape.scale = new Vector3(8f, 14f, 0f);

        var colorOverLifetime = ps.colorOverLifetime;
        colorOverLifetime.enabled = true;
        var gradient = new Gradient();
        gradient.SetKeys(
            new[] { new GradientColorKey(Color.white, 0f), new GradientColorKey(Color.white, 1f) },
            new[] { new GradientAlphaKey(0f, 0f), new GradientAlphaKey(1f, 0.3f), new GradientAlphaKey(0f, 1f) }
        );
        colorOverLifetime.color = gradient;

        var sizeOverLifetime = ps.sizeOverLifetime;
        sizeOverLifetime.enabled = true;
        sizeOverLifetime.size = new ParticleSystem.MinMaxCurve(1f, new AnimationCurve(
            new Keyframe(0f, 0.5f), new Keyframe(1f, 1.2f)
        ));

        // Use default particle material (additive)
        var renderer = go.GetComponent<ParticleSystemRenderer>();
        renderer.material = new Material(Shader.Find("Particles/Standard Unlit"));
        renderer.material.SetFloat("_Mode", 1); // Additive

        return ps;
    }
}
```

### `Assets/Scripts/BalloonGame/UI/Screens/SettingsPanel.cs`

```csharp
using System.Collections;
using UnityEngine;
using UnityEngine.UI;
using TMPro;

/// <summary>
/// Settings overlay panel. Slides in from the right or fades in.
/// Controls music volume, SFX volume, and haptics toggle.
/// Can be opened from TitleScreen or PauseManager.
/// </summary>
public class SettingsPanel : MonoBehaviour
{
    [Header("Panel")]
    [SerializeField] private RectTransform _panelRect;
    [SerializeField] private CanvasGroup _backdropGroup;
    [SerializeField] private CanvasGroup _panelGroup;

    [Header("Controls")]
    [SerializeField] private Slider _musicSlider;
    [SerializeField] private Slider _sfxSlider;
    [SerializeField] private Toggle _hapticsToggle;

    [Header("Labels")]
    [SerializeField] private TMP_Text _titleLabel;
    [SerializeField] private TMP_Text _musicLabel;
    [SerializeField] private TMP_Text _sfxLabel;
    [SerializeField] private TMP_Text _hapticsLabel;

    [Header("Buttons")]
    [SerializeField] private Button _closeButton;

    /// <summary>Whether the settings panel is currently visible.</summary>
    public bool IsOpen { get; private set; }

    private Coroutine _animationCoroutine;
    private float _panelHiddenX;
    private float _panelShownX;

    private void Awake()
    {
        // Panel slides in from right. Calculate positions based on panel width.
        if (_panelRect != null)
        {
            float panelWidth = _panelRect.rect.width;
            _panelShownX = 0f;
            _panelHiddenX = panelWidth + 50f; // Offscreen right with margin
        }

        SetupLabels();
        gameObject.SetActive(false);
    }

    private void OnEnable()
    {
        LoadSettings();
        BindListeners();
    }

    private void OnDisable()
    {
        UnbindListeners();
    }

    private void SetupLabels()
    {
        if (_titleLabel != null)
        {
            _titleLabel.text = "SETTINGS";
            _titleLabel.fontSize = 48;
            _titleLabel.fontStyle = FontStyles.Bold;
            _titleLabel.color = UIColors.TextPrimary;
        }

        if (_musicLabel != null)
        {
            _musicLabel.text = "MUSIC";
            _musicLabel.fontSize = 28;
            _musicLabel.color = UIColors.TextSecondary;
        }

        if (_sfxLabel != null)
        {
            _sfxLabel.text = "SFX";
            _sfxLabel.fontSize = 28;
            _sfxLabel.color = UIColors.TextSecondary;
        }

        if (_hapticsLabel != null)
        {
            _hapticsLabel.text = "HAPTICS";
            _hapticsLabel.fontSize = 28;
            _hapticsLabel.color = UIColors.TextSecondary;
        }
    }

    private void LoadSettings()
    {
        if (SaveManager.Instance == null) return;

        var data = SaveManager.Instance.Data;

        if (_musicSlider != null)
        {
            _musicSlider.SetValueWithoutNotify(data.musicVolume);
        }

        if (_sfxSlider != null)
        {
            _sfxSlider.SetValueWithoutNotify(data.sfxVolume);
        }

        if (_hapticsToggle != null)
        {
            _hapticsToggle.SetIsOnWithoutNotify(data.hapticsEnabled);
        }
    }

    private void BindListeners()
    {
        if (_musicSlider != null) _musicSlider.onValueChanged.AddListener(OnMusicChanged);
        if (_sfxSlider != null) _sfxSlider.onValueChanged.AddListener(OnSFXChanged);
        if (_hapticsToggle != null) _hapticsToggle.onValueChanged.AddListener(OnHapticsChanged);
        if (_closeButton != null) _closeButton.onClick.AddListener(Close);
    }

    private void UnbindListeners()
    {
        if (_musicSlider != null) _musicSlider.onValueChanged.RemoveListener(OnMusicChanged);
        if (_sfxSlider != null) _sfxSlider.onValueChanged.RemoveListener(OnSFXChanged);
        if (_hapticsToggle != null) _hapticsToggle.onValueChanged.RemoveListener(OnHapticsChanged);
        if (_closeButton != null) _closeButton.onClick.RemoveListener(Close);
    }

    private void OnMusicChanged(float value)
    {
        if (SaveManager.Instance == null) return;
        SaveManager.Instance.Data.musicVolume = value;

        var audioManager = FindAnyObjectByType<AudioManager>();
        if (audioManager != null)
        {
            audioManager.MusicVolume = value;
        }

        SaveManager.Instance.Save();
    }

    private void OnSFXChanged(float value)
    {
        if (SaveManager.Instance == null) return;
        SaveManager.Instance.Data.sfxVolume = value;

        var audioManager = FindAnyObjectByType<AudioManager>();
        if (audioManager != null)
        {
            audioManager.SFXVolume = value;
        }

        SaveManager.Instance.Save();
    }

    private void OnHapticsChanged(bool enabled)
    {
        if (SaveManager.Instance == null) return;
        SaveManager.Instance.Data.hapticsEnabled = enabled;
        SaveManager.Instance.Save();
    }

    /// <summary>
    /// Open the settings panel with slide-in animation.
    /// </summary>
    public void Open()
    {
        if (IsOpen) return;
        IsOpen = true;
        gameObject.SetActive(true);

        if (_animationCoroutine != null) StopCoroutine(_animationCoroutine);
        _animationCoroutine = StartCoroutine(AnimateOpen());
    }

    /// <summary>
    /// Close the settings panel with slide-out animation.
    /// </summary>
    public void Close()
    {
        if (!IsOpen) return;

        if (_animationCoroutine != null) StopCoroutine(_animationCoroutine);
        _animationCoroutine = StartCoroutine(AnimateClose());
    }

    private IEnumerator AnimateOpen()
    {
        float duration = GameConstants.SETTINGS_SLIDE_DURATION;
        float elapsed = 0f;

        // Fade in backdrop
        if (_backdropGroup != null)
        {
            _backdropGroup.alpha = 0f;
            _backdropGroup.blocksRaycasts = true;
        }

        // Slide panel from right
        if (_panelRect != null)
        {
            var pos = _panelRect.anchoredPosition;
            pos.x = _panelHiddenX;
            _panelRect.anchoredPosition = pos;
        }

        while (elapsed < duration)
        {
            elapsed += Time.unscaledDeltaTime;
            float t = Mathf.SmoothStep(0f, 1f, Mathf.Clamp01(elapsed / duration));

            if (_backdropGroup != null)
            {
                _backdropGroup.alpha = t * 0.7f;
            }

            if (_panelRect != null)
            {
                var pos = _panelRect.anchoredPosition;
                pos.x = Mathf.Lerp(_panelHiddenX, _panelShownX, t);
                _panelRect.anchoredPosition = pos;
            }

            yield return null;
        }

        // Snap to final values
        if (_backdropGroup != null) _backdropGroup.alpha = 0.7f;
        if (_panelRect != null)
        {
            var pos = _panelRect.anchoredPosition;
            pos.x = _panelShownX;
            _panelRect.anchoredPosition = pos;
        }

        _animationCoroutine = null;
    }

    private IEnumerator AnimateClose()
    {
        float duration = GameConstants.SETTINGS_SLIDE_DURATION;
        float elapsed = 0f;

        while (elapsed < duration)
        {
            elapsed += Time.unscaledDeltaTime;
            float t = Mathf.SmoothStep(0f, 1f, Mathf.Clamp01(elapsed / duration));

            if (_backdropGroup != null)
            {
                _backdropGroup.alpha = 0.7f * (1f - t);
            }

            if (_panelRect != null)
            {
                var pos = _panelRect.anchoredPosition;
                pos.x = Mathf.Lerp(_panelShownX, _panelHiddenX, t);
                _panelRect.anchoredPosition = pos;
            }

            yield return null;
        }

        if (_backdropGroup != null)
        {
            _backdropGroup.alpha = 0f;
            _backdropGroup.blocksRaycasts = false;
        }

        IsOpen = false;
        gameObject.SetActive(false);
        _animationCoroutine = null;
    }
}
```

```csharp
using System;
using System.Collections;
using UnityEngine;
using UnityEngine.UI;
using TMPro;

/// <summary>
/// First-run tutorial overlay. Shows 3 instructional steps over a dark overlay.
/// Checks SaveData.hasCompletedTutorial; skips entirely if already completed.
/// Tap anywhere to advance through steps.
/// </summary>
public class TutorialOverlay : MonoBehaviour
{
    [Header("Overlay")]
    [SerializeField] private CanvasGroup _overlayGroup;
    [SerializeField] private Image _overlayBackground;

    [Header("Instruction Elements")]
    [SerializeField] private TMP_Text _instructionText;
    [SerializeField] private TMP_Text _tapToContinueText;
    [SerializeField] private RectTransform _arrowIndicator;

    /// <summary>Fired when the tutorial completes or is skipped.</summary>
    public event Action OnTutorialComplete;

    private int _currentStep;
    private const int TotalSteps = 3;
    private bool _isActive;
    private bool _canAdvance;

    private static readonly TutorialStepData[] Steps = new[]
    {
        new TutorialStepData
        {
            InstructionText = "DRAG TO AIM",
            // Arrow points toward launch area (lower-center of screen)
            ArrowPosition = new Vector2(0f, -500f),
            ArrowRotation = 0f
        },
        new TutorialStepData
        {
            InstructionText = "RELEASE TO THROW",
            // Same area, different instruction
            ArrowPosition = new Vector2(0f, -400f),
            ArrowRotation = 0f
        },
        new TutorialStepData
        {
            InstructionText = "POP TO SCORE",
            // Arrow points toward balloon area (upper-center of screen)
            ArrowPosition = new Vector2(0f, 300f),
            ArrowRotation = 180f
        }
    };

    private struct TutorialStepData
    {
        public string InstructionText;
        public Vector2 ArrowPosition;
        public float ArrowRotation;
    }

    /// <summary>
    /// Attempt to show the tutorial. Returns false if already completed.
    /// </summary>
    public bool TryShow()
    {
        if (SaveManager.Instance != null && SaveManager.Instance.Data.hasCompletedTutorial)
        {
            OnTutorialComplete?.Invoke();
            return false;
        }

        Show();
        return true;
    }

    /// <summary>
    /// Force-show the tutorial regardless of completion state.
    /// </summary>
    public void Show()
    {
        gameObject.SetActive(true);
        _isActive = true;
        _currentStep = 0;

        if (_overlayGroup != null)
        {
            _overlayGroup.alpha = 0f;
        }

        StartCoroutine(FadeInAndShowStep());
    }

    private void Update()
    {
        if (!_isActive || !_canAdvance) return;

        bool tapped = false;

        if (Input.touchCount > 0 && Input.GetTouch(0).phase == TouchPhase.Began)
        {
            tapped = true;
        }
        else if (Input.GetMouseButtonDown(0))
        {
            tapped = true;
        }

        if (tapped)
        {
            AdvanceStep();
        }
    }

    private void AdvanceStep()
    {
        _currentStep++;

        if (_currentStep >= TotalSteps)
        {
            CompleteTutorial();
        }
        else
        {
            ShowStep(_currentStep);
        }
    }

    private IEnumerator FadeInAndShowStep()
    {
        // Fade in the overlay
        float duration = 0.3f;
        float elapsed = 0f;

        if (_overlayBackground != null)
        {
            _overlayBackground.color = new Color(0f, 0f, 0f, 0f);
        }

        while (elapsed < duration)
        {
            elapsed += Time.unscaledDeltaTime;
            float t = Mathf.Clamp01(elapsed / duration);

            if (_overlayGroup != null)
            {
                _overlayGroup.alpha = t;
            }

            if (_overlayBackground != null)
            {
                _overlayBackground.color = new Color(0f, 0f, 0f, t * GameConstants.TUTORIAL_OVERLAY_ALPHA);
            }

            yield return null;
        }

        ShowStep(0);
    }

    private void ShowStep(int stepIndex)
    {
        if (stepIndex < 0 || stepIndex >= Steps.Length) return;

        var step = Steps[stepIndex];
        _canAdvance = false;

        if (_instructionText != null)
        {
            _instructionText.text = step.InstructionText;
            _instructionText.fontSize = 56;
            _instructionText.fontStyle = FontStyles.Bold;
            _instructionText.alignment = TextAlignmentOptions.Center;
            _instructionText.color = UIColors.NeonCyan;
        }

        if (_arrowIndicator != null)
        {
            _arrowIndicator.anchoredPosition = step.ArrowPosition;
            _arrowIndicator.localRotation = Quaternion.Euler(0f, 0f, step.ArrowRotation);
            _arrowIndicator.gameObject.SetActive(true);
        }

        if (_tapToContinueText != null)
        {
            _tapToContinueText.text = stepIndex < TotalSteps - 1 ? "TAP TO CONTINUE" : "TAP TO BEGIN";
            _tapToContinueText.fontSize = 24;
            _tapToContinueText.color = UIColors.TextSecondary;
            _tapToContinueText.alignment = TextAlignmentOptions.Center;
        }

        // Brief delay before allowing advancement to prevent accidental double-tap
        StartCoroutine(EnableAdvanceAfterDelay(0.4f));
    }

    private IEnumerator EnableAdvanceAfterDelay(float delay)
    {
        float elapsed = 0f;
        while (elapsed < delay)
        {
            elapsed += Time.unscaledDeltaTime;
            yield return null;
        }
        _canAdvance = true;
    }

    private void CompleteTutorial()
    {
        _isActive = false;
        _canAdvance = false;

        if (SaveManager.Instance != null)
        {
            SaveManager.Instance.Data.hasCompletedTutorial = true;
            SaveManager.Instance.Save();
        }

        StartCoroutine(FadeOutAndClose());
    }

    private IEnumerator FadeOutAndClose()
    {
        float duration = 0.3f;
        float elapsed = 0f;

        while (elapsed < duration)
        {
            elapsed += Time.unscaledDeltaTime;
            float t = Mathf.Clamp01(elapsed / duration);

            if (_overlayGroup != null)
            {
                _overlayGroup.alpha = 1f - t;
            }

            yield return null;
        }

        OnTutorialComplete?.Invoke();
        gameObject.SetActive(false);
    }
}
```

```csharp
/// <summary>
/// In-game pause system. Provides a HUD pause button and a pause menu overlay

{
    [Header("HUD")]
    [SerializeField] private Button _pauseButton;

    [Header("Pause Menu")]
    [SerializeField] private GameObject _pauseMenuRoot;
    [SerializeField] private CanvasGroup _pauseMenuGroup;
    [SerializeField] private Button _resumeButton;
    [SerializeField] private Button _settingsButton;
    [SerializeField] private Button _quitButton;

    [SerializeField] private GameObject _quitConfirmRoot;
    [SerializeField] private CanvasGroup _quitConfirmGroup;
    [SerializeField] private TMP_Text _quitConfirmText;
    [SerializeField] private Button _quitConfirmYes;
    [SerializeField] private Button _quitConfirmNo;

    [Header("Labels")]
    [SerializeField] private TMP_Text _pauseTitle;
    [SerializeField] private TMP_Text _resumeLabel;
    [SerializeField] private TMP_Text _settingsLabel;
    [SerializeField] private TMP_Text _quitLabel;

    [SerializeField] private SettingsPanel _settingsPanel;

    private void Awake()

        if (_quitConfirmRoot != null) _quitConfirmRoot.SetActive(false);
    }

    private void OnEnable()

        if (_resumeButton != null) _resumeButton.onClick.AddListener(Resume);
        if (_settingsButton != null) _settingsButton.onClick.AddListener(OpenSettings);
        if (_quitButton != null) _quitButton.onClick.AddListener(ShowQuitConfirmation);
        if (_quitConfirmYes != null) _quitConfirmYes.onClick.AddListener(ConfirmQuit);
        if (_quitConfirmNo != null) _quitConfirmNo.onClick.AddListener(CancelQuit);

    {
        if (_pauseButton != null) _pauseButton.onClick.RemoveListener(TogglePause);
        if (_resumeButton != null) _resumeButton.onClick.RemoveListener(Resume);
        if (_settingsButton != null) _settingsButton.onClick.RemoveListener(OpenSettings);
        if (_quitButton != null) _quitButton.onClick.RemoveListener(ShowQuitConfirmation);
        if (_quitConfirmYes != null) _quitConfirmYes.onClick.RemoveListener(ConfirmQuit);
        if (_quitConfirmNo != null) _quitConfirmNo.onClick.RemoveListener(CancelQuit);

        // Safety: restore timeScale if disabled while paused
        if (IsPaused)
        {
            Time.timeScale = _previousTimeScale;
            IsPaused = false;
        }
    }

    private void SetupLabels()
    {
        if (_pauseTitle != null)
        {
            _pauseTitle.text = "PAUSED";
            _pauseTitle.fontSize = 64;
            _pauseTitle.fontStyle = FontStyles.Bold;
            _pauseTitle.alignment = TextAlignmentOptions.Center;
            _pauseTitle.color = UIColors.TextPrimary;

        if (_resumeLabel != null)

            _resumeLabel.text = "RESUME";
            _resumeLabel.fontSize = 36;
            _resumeLabel.color = UIColors.TextPrimary;

        if (_settingsLabel != null)

            _settingsLabel.text = "SETTINGS";
            _settingsLabel.fontSize = 36;
            _settingsLabel.color = UIColors.TextPrimary;

        if (_quitLabel != null)

            _quitLabel.text = "QUIT RUN";
            _quitLabel.fontSize = 36;
            _quitLabel.color = UIColors.NeonCyan;

        if (_quitConfirmText != null)

            _quitConfirmText.text = "ABANDON THIS RUN?";
            _quitConfirmText.fontSize = 40;
            _quitConfirmText.fontStyle = FontStyles.Bold;
            _quitConfirmText.alignment = TextAlignmentOptions.Center;
            _quitConfirmText.color = UIColors.TextPrimary;
        }

    /// <summary>Toggle pause state.</summary>
    public void TogglePause()
    {
        if (IsPaused)
        {
            Resume();
        }
        else
        {
            Pause();
        }
    }

    /// <summary>Pause the game.</summary>
    public void Pause()
    {
        if (IsPaused) return;

        IsPaused = true;
        _previousTimeScale = Time.timeScale;
        Time.timeScale = 0f;

        if (_pauseMenuRoot != null)
        {
            _pauseMenuRoot.SetActive(true);
        }

        if (_pauseMenuGroup != null)
        {
            StartCoroutine(FadeCanvasGroup(_pauseMenuGroup, 0f, 1f, 0.15f));
        }

        // Hide pause button while menu is showing
        if (_pauseButton != null)
        {
            _pauseButton.gameObject.SetActive(false);
        }
    }

    /// <summary>Resume the game.</summary>
    public void Resume()
    {
        if (!IsPaused) return;

        // Close settings if open
        if (_settingsPanel != null && _settingsPanel.IsOpen)
        {
            _settingsPanel.Close();
        }

        // Hide quit confirmation if visible
        if (_quitConfirmRoot != null)
        {
            _quitConfirmRoot.SetActive(false);
        }

        IsPaused = false;
        Time.timeScale = _previousTimeScale;

        if (_pauseMenuRoot != null)
        {
            _pauseMenuRoot.SetActive(false);
        }

        if (_pauseMenuGroup != null)
        {
            _pauseMenuGroup.alpha = 0f;
        }

        // Restore pause button
        if (_pauseButton != null)
        {
            _pauseButton.gameObject.SetActive(true);
        }
    }

    private void OpenSettings()
    {
        if (_settingsPanel != null)
        {
            _settingsPanel.Open();
        }
    }

    private void ShowQuitConfirmation()
    {
        if (_quitConfirmRoot != null)
        {
            _quitConfirmRoot.SetActive(true);
        }

        if (_quitConfirmGroup != null)
        {
            StartCoroutine(FadeCanvasGroup(_quitConfirmGroup, 0f, 1f, 0.15f));
        }
    }

    private void CancelQuit()
    {
        if (_quitConfirmRoot != null)
        {
            _quitConfirmRoot.SetActive(false);
        }

        if (_quitConfirmGroup != null)
        {
            _quitConfirmGroup.alpha = 0f;
        }
    }

    private void ConfirmQuit()
    {
        // Restore timeScale before transitioning
        Time.timeScale = _previousTimeScale;
        IsPaused = false;

        if (_pauseMenuRoot != null) _pauseMenuRoot.SetActive(false);
        if (_quitConfirmRoot != null) _quitConfirmRoot.SetActive(false);

        ScreenTransition.FadeTo(() =>
        {
            // Return to title screen
            if (_titleScreen != null)
            {
                _titleScreen.Show();
            }

            // Hide pause button when on title screen
            if (_pauseButton != null)
            {
                _pauseButton.gameObject.SetActive(false);
            }
        });
    }

    /// <summary>
    /// Fade a CanvasGroup using unscaledDeltaTime (works while timeScale is 0).
    /// </summary>
    private IEnumerator FadeCanvasGroup(CanvasGroup group, float from, float to, float duration)
    {
        group.alpha = from;
        group.blocksRaycasts = true;
        float elapsed = 0f;

        while (elapsed < duration)
        {
            elapsed += Time.unscaledDeltaTime;
            group.alpha = Mathf.Lerp(from, to, Mathf.Clamp01(elapsed / duration));
            yield return null;
        }

        group.alpha = to;
        group.blocksRaycasts = to > 0.01f;
    }
}
```

## Notes

- **ScreenTransition replaces FadeOverlay for global transitions.** FadeOverlay from Plan 04 is a per-Canvas component. ScreenTransition is a lazy singleton that works from anywhere without scene setup. Both can coexist — FadeOverlay for in-run room transitions, ScreenTransition for menu-level transitions.
- **Single SettingsPanel instance** is shared between TitleScreen and PauseManager. Both hold a serialized reference to the same GameObject.
- **TutorialOverlay fires after the first TitleScreen tap.** The recommended flow: TitleScreen tap → ScreenTransition fade → activate gameplay → TutorialOverlay.TryShow() → if first run, show tutorial before unpausing.
- **PauseManager uses `Time.unscaledDeltaTime`** for all UI animations so they work while `timeScale` is 0.
- **All text uses TextMeshPro.** No legacy Unity Text components.
- **All colors reference `UIColors` static class.** No hardcoded color values in MonoBehaviours (except the mist particle system which uses raw values for particle colors that don't map to UI color constants).
- **SaveData.hasCompletedTutorial** is the only schema change. The field defaults to `false` and is backward-compatible with existing save files (JsonUtility assigns default values to missing fields).
